1. Why Rate Limit on the Client?

// Servers already enforce rate limits (429 Too Many Requests)
// But waiting for a 429 is wasteful:
//   1. Wastes bandwidth (full request/response round-trip)
//   2. Adds latency (user waits for the 429 before retry)
//   3. Wastes server resources (server must parse & reject)
//   4. Degrades UX (user sees "too many requests" error)
//
// Client-side rate limiting sits between UI and network:
//
// Component → useQuery → API Layer → [Rate Limiter] → Network → Server
//                                        ↓
//                                   queue / drop
//                                   if over budget
//
// It's transparent to components — they call api.fetch() as normal
// The rate limiter intercepts and queues excess calls

3. Token Bucket — How It Works

// Imagine a bucket that holds 10 tokens (capacity = 10)
// Tokens refill at 2 per second (refillRate = 2/s)
//
// t=0s:  bucket = 10 tokens (full)
// Page load fires 8 requests → 8 tokens consumed → 2 remaining
// t=0.5s: bucket refills 1 token → 3 available
// t=1s:   bucket refills 1 more → 4 available
// User types fast → 4 requests → 0 tokens
// Next request → BLOCKED until token available (0.5s wait)
//
// Key insight: burst capacity (10) lets page loads work
// Sustained rate (2/s) prevents overwhelming the server
class TokenBucket {
  constructor(capacity, refillRate) {
    this.capacity = capacity;
    this.tokens = capacity;       // Start full
    this.refillRate = refillRate; // Tokens per ms
    this.lastRefill = Date.now();
  }

  refill() {
    const elapsed = Date.now() - this.lastRefill;
    this.tokens = Math.min(
      this.capacity,
      this.tokens + elapsed * this.refillRate
    );
    this.lastRefill = Date.now();
  }

  consume() {
    this.refill();
    if (this.tokens >= 1) {
      this.tokens -= 1;
      return true;  // Allowed
    }
    return false;    // Rate limited
  }

  get waitTime() {
    this.refill();
    if (this.tokens >= 1) return 0;
    return Math.ceil((1 - this.tokens) / this.refillRate);
  }
}

4. Per-Endpoint Configuration

// Different endpoints have different tolerances:
//
// Search autocomplete → restrictive (debounced anyway)
//   burst: 3, rate: 0.5/s → max 3 immediate, then 1 every 2s
//
// Analytics events → batch-friendly, low priority
//   burst: 5, rate: 0.2/s → batch and drain slowly
//
// Product API → generous (user browsing)
//   burst: 10, rate: 2/s → allows page load burst
//
// Background sync → very restrictive
//   burst: 1, rate: 0.1/s → 1 request every 10 seconds

const endpointConfigs = {
  '/api/search':    { burst: 3,  rate: 0.5 / 1000 },
  '/api/analytics': { burst: 5,  rate: 0.2 / 1000 },
  '/api/products':  { burst: 10, rate: 2 / 1000 },
  '/api/sync':      { burst: 1,  rate: 0.1 / 1000 },
};

1. Rate-Limited API Client with Queue

class RateLimitedClient {
  constructor() {
    this.buckets = new Map();
    this.queues = new Map();
  }

  getBucket(endpoint) {
    if (!this.buckets.has(endpoint)) {
      const config = endpointConfigs[endpoint] ?? { burst: 10, rate: 2 / 1000 };
      this.buckets.set(endpoint, new TokenBucket(config.burst, config.rate));
    }
    return this.buckets.get(endpoint);
  }

  async request(endpoint, fetchFn, { queue = true, priority = 0 } = {}) {
    const bucket = this.getBucket(endpoint);

    if (bucket.consume()) {
      return fetchFn();  // Token available → execute immediately
    }

    if (!queue) {
      throw new Error(`Rate limited: ${endpoint} (wait ${bucket.waitTime}ms)`);
    }

    // Queue the request for later execution
    return new Promise((resolve, reject) => {
      const q = this.queues.get(endpoint) ?? [];
      q.push({ fetchFn, resolve, reject, priority });
      q.sort((a, b) => b.priority - a.priority);
      this.queues.set(endpoint, q);
      this.scheduleProcessing(endpoint);
    });
  }

  scheduleProcessing(endpoint) {
    const bucket = this.getBucket(endpoint);
    const delay = bucket.waitTime || 100;

    setTimeout(() => {
      const q = this.queues.get(endpoint);
      if (!q?.length) return;
      if (bucket.consume()) {
        const item = q.shift();
        item.fetchFn().then(item.resolve).catch(item.reject);
      }
      if (q.length) this.scheduleProcessing(endpoint);
    }, delay);
  }
}

2. Visibility API — Pause When Tab Hidden

// When user switches tabs, pause background requests
// Real-world: dashboard polling every 5s — waste when tab hidden

class VisibilityAwareRateLimiter {
  constructor(client) {
    this.client = client;
    this.paused = false;
    this.pendingOnResume = [];

    document.addEventListener('visibilitychange', () => {
      if (document.hidden) {
        this.paused = true;
      } else {
        this.paused = false;
        // Flush pending requests (staggered)
        this.pendingOnResume.forEach((fn, i) => {
          setTimeout(fn, i * 200);  // 200ms apart to avoid burst
        });
        this.pendingOnResume = [];
      }
    });
  }

  async request(endpoint, fetchFn) {
    if (this.paused) {
      return new Promise((resolve, reject) => {
        this.pendingOnResume.push(() =>
          this.client.request(endpoint, fetchFn)
            .then(resolve).catch(reject)
        );
      });
    }
    return this.client.request(endpoint, fetchFn);
  }
}

3. Adaptive Rate Limiting

// Dynamically adjust rate based on server responses

class AdaptiveRateLimiter {
  constructor(bucket) {
    this.bucket = bucket;
    this.originalRate = bucket.refillRate;
  }

  onResponse(response) {
    // Read server's rate limit headers
    const remaining = response.headers.get('X-RateLimit-Remaining');
    const limit = response.headers.get('X-RateLimit-Limit');

    if (remaining !== null && limit !== null) {
      const ratio = parseInt(remaining) / parseInt(limit);

      if (ratio < 0.1) {
        // Less than 10% remaining → slow to 25% of normal rate
        this.bucket.refillRate = this.originalRate * 0.25;
      } else if (ratio < 0.3) {
        // Less than 30% → slow to 50%
        this.bucket.refillRate = this.originalRate * 0.5;
      } else {
        // Plenty of headroom → normal rate
        this.bucket.refillRate = this.originalRate;
      }
    }

    if (response.status === 429) {
      // Server rejected us → aggressively back off
      this.bucket.refillRate = this.originalRate * 0.1;
      this.bucket.tokens = 0;
    }
  }
}

FAANG Interview Questions

Q1: Why implement rate limiting on the client when the server already enforces it?
Client-side rate limiting is a proactive optimization, not redundancy. Waiting for a server 429 wastes a full round-trip (DNS + TCP + TLS + request + response), adds visible latency, and still consumes server resources to parse and reject. Client-side limiting prevents requests from ever leaving the browser, reducing bandwidth and server load. It also improves UX — instead of showing "429 Too Many Requests" errors, we queue requests transparently. In a real-time dashboard polling every second, client-side limiting can reduce server load by 80% when the tab is hidden (Visibility API). It's the difference between a polite client and an abusive one.
Follow-up: How do you sync client-side limits with server-side limits so they don't diverge?
Q2: Compare Token Bucket vs Leaky Bucket for frontend use.
Token Bucket is ideal for frontend because it allows controlled bursting. When a page loads, it might need 8 API calls simultaneously — Token Bucket's capacity parameter permits this burst, then limits sustained rate. Leaky Bucket forces a fixed drain rate with no burst allowance, which means page load requests would be queued and delayed unnecessarily. The trade-off: Token Bucket has variable output rate (bursty), Leaky Bucket has smooth output rate. For frontend, bursty is better — users expect instant page loads. For rate limiting uploads or streaming, Leaky Bucket's smoothness is preferable. I'd use Token Bucket for 95% of frontend rate limiting, Leaky Bucket for file upload progress throttling.
Follow-up: How would you implement a Token Bucket that supports different costs per request type?
Q3: How would you handle rate limiting in a micro-frontend architecture?
In micro-frontends, each independently deployed app could have its own rate limiter, leading to aggregate over-limiting or under-limiting. I'd centralize rate limiting in a shared API layer — either a Service Worker that intercepts all fetch requests from all micro-frontends, or a shared singleton rate limiter exposed via module federation. The shared limiter tracks a global budget per endpoint, ensuring the aggregate rate stays within server limits. Each micro-frontend registers its priority level, and the limiter allocates budget accordingly — checkout gets higher priority than recommendations. The Service Worker approach is cleanest because it's transparent and requires no coordination between micro-frontends.
Follow-up: What about rate limiting WebSocket messages in addition to HTTP?
Q4: How do you handle rate limiting with the Page Visibility API?
When a tab is hidden (document.hidden = true), I pause non-critical requests entirely and reduce critical polling frequency. For example, a dashboard polling every 5 seconds pauses background sync when hidden, queuing requests. When the tab becomes visible again, I don't flush all queued requests at once (that would burst-load the server) — instead I stagger them 200ms apart or just fetch the latest data once, discarding stale queued requests. Chrome already throttles setTimeout to 1000ms minimum for hidden tabs, but that still allows 1 request per second — explicit pausing is better. I also reduce WebSocket message processing for hidden tabs, buffering updates and applying them on resume.
Follow-up: How does this interact with Service Workers that may still be active?

1. Production Token Bucket with Priority Queue

class PriorityRateLimiter {
  constructor(capacity = 10, refillPerSec = 2) {
    this.bucket = new TokenBucket(capacity, refillPerSec / 1000);
    this.queue = [];
    this.processing = false;
  }

  execute(fetchFn, priority = 0) {
    return new Promise((resolve, reject) => {
      if (this.bucket.consume()) {
        fetchFn().then(resolve).catch(reject);
        return;
      }
      this.queue.push({ fetchFn, resolve, reject, priority });
      this.queue.sort((a, b) => b.priority - a.priority);
      this.drain();
    });
  }

  drain() {
    if (this.processing) return;
    this.processing = true;

    const tick = () => {
      if (!this.queue.length) {
        this.processing = false;
        return;
      }
      if (this.bucket.consume()) {
        const item = this.queue.shift();
        item.fetchFn().then(item.resolve).catch(item.reject);
      }
      setTimeout(tick, this.bucket.waitTime || 50);
    };
    tick();
  }
}

// Usage:
const limiter = new PriorityRateLimiter(10, 2);
limiter.execute(() => fetch('/api/products'), 10); // high priority
limiter.execute(() => fetch('/api/analytics'), 1); // low priority

2. Reading Server Rate Limit Headers

// Standard rate limit headers from server:
// X-RateLimit-Limit: 100      → max requests in window
// X-RateLimit-Remaining: 23   → requests left in window
// X-RateLimit-Reset: 1618884480 → unix timestamp when window resets
// Retry-After: 30             → seconds to wait (on 429)

function parseRateLimitHeaders(response) {
  return {
    limit: parseInt(response.headers.get('X-RateLimit-Limit')) || null,
    remaining: parseInt(response.headers.get('X-RateLimit-Remaining')) || null,
    resetAt: parseInt(response.headers.get('X-RateLimit-Reset')) * 1000 || null,
    retryAfter: parseInt(response.headers.get('Retry-After')) || null,
  };
}

// Use these to configure client-side bucket dynamically:
function syncBucketWithServer(bucket, headers) {
  if (headers.remaining !== null && headers.limit !== null) {
    bucket.tokens = Math.min(headers.remaining, bucket.capacity);
  }
}

Decision Tree

Need to allow page-load burst?
├── Yes → Token Bucket (capacity = expected burst size)
└── No
    ├── Need smooth output rate? → Leaky Bucket
    └── Need simple counter? → Fixed Window

Need per-endpoint limits? → Map<endpoint, TokenBucket>

Tab hidden? → Pause optional requests (Visibility API)

Server returning rate headers? → Sync bucket state adaptively

Production Checklist

✅ Use Token Bucket for frontend (allows controlled bursting)
✅ Configure per-endpoint rate limits (search ≠ analytics)
✅ Implement request queuing with priority (checkout > analytics)
✅ Pause non-critical requests when tab is hidden
✅ Read server's X-RateLimit-* headers to sync state
✅ On 429: respect Retry-After header, reduce bucket rate
✅ Stagger queued requests on tab resume (no burst-flush)
✅ Log rate limit events to monitoring (detect misconfiguration)
✅ Set queue size limits (don't queue 1000 requests)
✅ Combine with circuit breaker for sustained outages
