Effects
Listen for actions, perform side effects (HTTP, router, localStorage), dispatch new action with result. Use RxJS operators (switchMap, mergeMap, catchError).
Selectors
Memoised functions to derive slices of state. createSelector composes selectors. Components subscribe via store.select() — only re-render when derived value changes.
EntityAdapter
Manages collections in a normalised shape: { ids: string[], entities: Map }. Built-in CRUD adapter methods: addOne, removeOne, updateOne, upsertMany.

Selectors (createSelector) and EntityAdapter

// ─── createSelector ───
const selectOrders       = (state: AppState) => state.orders.entities;
const selectOrderIds     = (state: AppState) => state.orders.ids;
const selectStatusFilter = (state: AppState) => state.ui.statusFilter;

const selectFilteredOrders = createSelector(
  selectOrders,
  selectOrderIds,
  selectStatusFilter,
  (entities, ids, filter) =>           // memoised!
    ids
      .map(id => entities[id]!)
      .filter(o => o.status === filter)
);

// ─── EntityAdapter ───
const ordersAdapter = createEntityAdapter<Order>({
  selectId: o => o.id,
  sortComparer: (a, b) => a.date.localeCompare(b.date),
});

const ordersReducer = createReducer(
  ordersAdapter.getInitialState({ loading: false, error: null }),
  on(loadOrdersSuccess, (state, { orders }) =>
    ordersAdapter.setAll(orders, { ...state, loading: false })   // replaces all
  ),
  on(approveOrderSuccess, (state, { id }) =>
    ordersAdapter.updateOne({ id, changes: { status: 'approved' } }, state)
  ),
);
export const { selectAll, selectById } = ordersAdapter.getSelectors();

NgRx Effects — full type-safe pattern

@Injectable()
export class OrderEffects {
  loadOrders$ = createEffect(() =>
    this.actions$.pipe(
      ofType(OrderActions.loadOrders),
      switchMap(() =>
        this.orderService.getAll().pipe(
          map(orders  => OrderActions.loadOrdersSuccess({ orders })),
          catchError(err => of(OrderActions.loadOrdersFailure({ error: err.message })))
        )
      )
    )
  );

  approveOrder$ = createEffect(() =>
    this.actions$.pipe(
      ofType(OrderActions.approveOrder),
      mergeMap(({ id }) =>             // mergeMap: allow concurrent approvals
        this.orderService.approve(id).pipe(
          map(() => OrderActions.approveOrderSuccess({ id })),
          catchError(err => of(OrderActions.approveOrderFailure({ error: err.message })))
        )
      )
    )
  );
}
