Type-safe permission system in TypeScript

// Template literal type — namespace:action format
type Namespace = 'document' | 'user' | 'billing';
type Action = 'read' | 'write' | 'delete' | 'manage' | 'invite';
type Permission = `${Namespace}:${Action}`;  // compile-time safety ✓

const ROLE_PERMISSIONS = {
  viewer:  ['document:read', 'user:read'] as Permission[],
  editor:  ['document:read', 'document:write', 'user:read'] as Permission[],
  admin:   ['document:read', 'document:write', 'document:delete',
            'user:read', 'user:invite', 'billing:manage'] as Permission[],
} satisfies Record<string, Permission[]>;

// Resolve effective permissions (roles + overrides)
function resolvePermissions(user: { roles: (keyof typeof ROLE_PERMISSIONS)[]; overrides?: Permission[] }) {
  const result = new Set<Permission>();
  user.roles.forEach(r => ROLE_PERMISSIONS[r].forEach(p => result.add(p)));
  user.overrides?.forEach(p => result.add(p));
  return result;
}

// React hook (UX only — server enforces)
function usePermissions() {
  const { user } = useAuth();
  const perms = useMemo(() => resolvePermissions(user), [user.roles]);
  return { can: (p: Permission) => perms.has(p) };
}
