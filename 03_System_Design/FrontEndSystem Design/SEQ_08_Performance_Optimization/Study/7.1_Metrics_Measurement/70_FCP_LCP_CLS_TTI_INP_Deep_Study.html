FCP — First Contentful Paint (Browser Internals)

// FCP fires when the browser first renders ANY content to the screen
// "Content" = text, image, SVG, non-white canvas
//
// What happens internally:
// 1. HTML bytes arrive (after TTFB)
// 2. Parser tokenizes HTML incrementally
// 3. DOM tree construction begins
// 4. CSSOM built from <style> / linked stylesheets
//    ⚠️ CSS is RENDER-BLOCKING — no paint until CSSOM ready
// 5. Render tree = DOM + CSSOM (only visible nodes)
// 6. Layout phase — calculate geometry (x, y, width, height)
// 7. Paint phase — rasterize pixels to layers
// 8. Composite — GPU composites layers → pixels on screen
// 9. FCP fires at step 8 for the first meaningful render
//
// FCP Blockers:
// - Synchronous <script> in <head> (parser-blocking)
// - Large unoptimized CSS files (render-blocking)
// - Web fonts with font-display: block (text invisible until font loads)
// - Server-side delay (high TTFB eats into FCP budget)

LCP — Largest Contentful Paint (Element Selection Algorithm)

// LCP Algorithm (Chrome implementation):
//
// 1. On each render, browser identifies "contentful" paint candidates
// 2. Candidates: <img>, <video poster>, background-image, text blocks
// 3. Size = visible area (clipped to viewport), excluding padding/border
// 4. The LARGEST candidate at each render frame is a "LCP candidate"
// 5. LCP keeps updating — new larger elements can override previous
// 6. Finalization: LCP stops updating on first user interaction
//    (keydown, mousedown, pointerdown, or tap)
//
// Edge cases:
// - Image loading progressively: LCP fires when fully decoded
// - Lazy-loaded images: LCP = time image becomes visible, not load event
// - Text with web fonts: LCP fires when font renders (or fallback shows)
// - Carousel: First slide is LCP (before user interacts)
//
// Sub-parts of LCP (for optimization):
// LCP = TTFB + Resource Load Delay + Resource Load Time + Render Delay
//
// TTFB:               Server response time
// Resource Load Delay: Time from TTFB until browser starts loading LCP resource
// Resource Load Time:  Download time of the LCP resource
// Render Delay:        Time from resource loaded to actual paint

LCP Optimization Strategies

// Strategy 1: Minimize TTFB
// - CDN for static assets (CloudFront, Fastly, Cloudflare)
// - Edge rendering (Cloudflare Workers, Vercel Edge Runtime)
// - HTTP/3 with 0-RTT for repeat visits
// - Stale-while-revalidate cache strategy

// Strategy 2: Eliminate Resource Load Delay
<link rel="preload" as="image" href="/hero.avif" fetchpriority="high">
// Without preload: browser discovers image only when parsing HTML body
// With preload: browser starts downloading immediately from <head>

// Strategy 3: Minimize Resource Load Time
<img
  src="/hero.avif"
  srcset="/hero-400.avif 400w, /hero-800.avif 800w, /hero-1200.avif 1200w"
  sizes="(max-width: 600px) 100vw, 50vw"
  fetchpriority="high"
  decoding="async"
  alt="Hero image"
>

// Strategy 4: Minimize Render Delay
// - Inline critical CSS (<style> in <head>)
// - Defer non-critical JS (won't block rendering)
// - Avoid document.write() (forces parser restart)
// - Use font-display: swap (text visible during font load)

CLS Scoring Algorithm

// CLS = Cumulative Layout Shift
// Measures unexpected visual movement of page elements
//
// Individual shift score = Impact Fraction × Distance Fraction
//
// Impact Fraction:
//   Union of the element's visible area BEFORE and AFTER the shift
//   divided by viewport area
//
// Distance Fraction:
//   Maximum distance any element moved (horizontal or vertical)
//   divided by the larger viewport dimension
//
// Session Window approach (since Chrome 93):
// Group shifts into "windows" with max 5s duration, 1s gap
// CLS = the WORST session window score
// This prevents long-lived SPAs from accumulating CLS forever
//
// Example calculation:
// Banner (300×100px) at (0,0) shifts to (0,50)
// Viewport: 400×800
// Impact: (300×100 + 300×50) / (400×800) = 45000/320000 = 0.14
// Distance: 50/800 = 0.0625
// Score: 0.14 × 0.0625 = 0.00875

CLS Prevention Patterns

// 1. Always set dimensions on media
<img src="photo.webp" width="800" height="600" alt="Photo">

// 2. Use aspect-ratio for responsive images
img { aspect-ratio: 16 / 9; width: 100%; height: auto; }

// 3. Reserve space for ads
.ad-slot { min-height: 250px; contain: layout; }

// 4. Font size-adjust to match fallback metrics
@font-face {
  font-family: 'CustomFont';
  src: url('font.woff2');
  font-display: swap;
  size-adjust: 105%;  /* Match x-height of system fallback */
  ascent-override: 90%;
  descent-override: 20%;
}

// 5. Use transform for animations (no layout shift)
// ❌ Causes CLS:
.notification { top: 0; transition: top 0.3s; }
// ✅ No CLS:
.notification { transform: translateY(-100%); transition: transform 0.3s; }

// 6. content-visibility for below-fold content
.below-fold { content-visibility: auto; contain-intrinsic-size: 0 500px; }

INP — Full Interaction Latency Breakdown

// INP = Input Delay + Processing Time + Presentation Delay
//
// ┌──────────────┬──────────────────┬───────────────────┐
// │ Input Delay   │ Processing Time  │ Presentation Delay│
// │ (main thread  │ (your event      │ (style, layout,   │
// │  was busy)    │  handlers run)   │  paint, composite)│
// └──────────────┴──────────────────┴───────────────────┘
// ← User clicks                                → Next frame painted
//
// Input Delay — WHY is main thread busy?
// - Long task from another event handler
// - Timer callback (setTimeout/setInterval)
// - Third-party script executing
// - Framework hydration/reconciliation
//
// Processing Time — HOW long does your handler take?
// - State updates triggering synchronous renders
// - Expensive computations (filtering large arrays, sorting)
// - Synchronous DOM reads/writes causing forced reflow
//
// Presentation Delay — WHY is rendering slow?
// - Large DOM tree (>1500 nodes) → slow layout
// - Complex CSS selectors → slow style recalculation
// - Too many layers → slow compositing
// - No GPU acceleration for animations

INP Optimization Strategies

// 1. Yield to main thread to reduce input delay
function yieldToMain() {
  return new Promise(resolve => setTimeout(resolve, 0));
}

async function processItems(items) {
  for (let i = 0; i < items.length; i++) {
    processItem(items[i]);
    if (i % 50 === 0) await yieldToMain(); // Let browser handle input
  }
}

// 2. Use scheduler.yield() (Chrome 115+)
async function handleClick() {
  updateUI(); // Immediate visual feedback
  await scheduler.yield(); // Yield, but KEEP priority in queue
  heavyComputation(); // Resume after browser processes pending input
}

// 3. Use startViewTransition for perceived responsiveness
document.startViewTransition(() => {
  updateDOM(); // Browser captures before/after, animates smoothly
});

// 4. Move computation to Web Worker
const worker = new Worker('sort-worker.js');
worker.postMessage({ data: largeArray, sortBy: 'date' });
worker.onmessage = (e) => renderSortedList(e.data);

TTI vs TBT vs INP — Why Google Changed Metrics

MetricMeasuresProblemStatus
TTITime until 5s quiet windowOne late long task delays TTI by 5s+; not reflective of actual experienceDeprecated from CWV
FIDInput delay of FIRST interactionOnly measures delay, not processing; only first interaction; most sites pass easilyReplaced by INP (Mar 2024)
TBTTotal blocking time (sum of >50ms portions)Lab-only; no interaction needed; correlates with INP but not identicalLab proxy for INP
INPFull latency of worst interaction (p98)—✅ Current CWV
INP is harder to pass than FID. Google found only 65% of origins passed FID, but swapping to INP dropped that to 55%. INP catches slow interactions FID missed because it measures ALL interactions, not just the first.

FAANG Interview Questions

Q1: Walk me through what happens from the moment a user clicks a link to LCP firing.
Navigation starts → DNS resolution → TCP + TLS handshake → HTTP request sent → server processes request → TTFB (first byte arrives) → browser starts parsing HTML incrementally → discovers CSS/JS/image resources → builds DOM tree incrementally → builds CSSOM from CSS → render tree = visible DOM + computed styles → FCP fires when first content pixels are painted → browser continues loading resources → LCP element's resource (usually hero image) finishes loading → browser decodes image, lays out, paints, composites → LCP fires. LCP keeps updating if a larger element renders later; it finalizes on first user interaction. The four sub-parts to optimize: TTFB (server speed), resource load delay (preload hints), resource load time (image optimization), and render delay (critical CSS).
Follow-up: Why might LCP differ between lab and field measurements?
Q2: Your e-commerce site has CLS of 0.35. How do you diagnose which elements are shifting?
Three diagnostic approaches: (1) Chrome DevTools — open Performance panel, check "Layout Shifts" in the Experience row. Each shift shows the affected elements and their before/after rects. (2) web-vitals attribution build — the CLS callback includes entries with sources array, each source has node (the DOM element) and previousRect/currentRect. Send this data to analytics. (3) Layout Shift Debugger extension — visualizes shifts in real-time with colored overlays. Common culprits on e-commerce: product images without dimensions (biggest offender), late-loading price/rating widgets, ad slots expanding, web fonts causing FOUT. I'd fix by: adding width/height to all images, using aspect-ratio CSS, reserving min-height for dynamic widgets, using font-display: optional or size-adjust to match fallback font metrics.
Follow-up: How does the session window approach affect CLS for SPAs?
Q3: Explain INP and how it differs from FID. Why did Google make the switch?
FID measured only the input delay of the first interaction — the time between user's first click/tap and when the browser could start running the handler. It had three problems: (1) it only measured delay, not processing time or rendering time, so a handler that took 500ms to execute still got a good FID score; (2) it only measured the first interaction, missing slow interactions later in the session; (3) most sites easily passed FID because by the time users interact, the main thread is usually idle. INP fixes all three: it measures the full latency (input delay + processing + presentation delay) of every interaction throughout the page lifetime, then reports the 98th percentile worst interaction. This catches real-world problems like slow accordion opens, filter rerenders, and modal animations. Google found INP better correlates with user satisfaction metrics — abandonment rate, task completion, and perceived responsiveness.
Follow-up: How would you optimize INP for a complex data table with sorting/filtering?
Q4: Design a Core Web Vitals monitoring pipeline for a site serving 100M monthly visits.
I'd architect three layers. Sampling layer: instrument all pages with the web-vitals library, but sample at 10% to control data volume (10M data points/month). Each beacon includes: metric name, value, rating, attribution data (LCP element, INP target, CLS sources), URL, device type, connection type, geography, and a session ID. Collection layer: beacons hit an edge function (Cloudflare Worker) that validates, enriches with server-side data (A/B test variant, deployment version), and writes to a streaming pipeline (Kafka/Kinesis). Processing layer: stream processor aggregates into p50, p75, p95 per metric per URL per device per geography per deployment. Store in a time-series database (InfluxDB/TimescaleDB). Dashboard: real-time p75 trending by URL group, automatic regression detection (>10% change vs 7-day baseline triggers PagerDuty), and deployment correlation (overlay deploy markers on metric timeline to catch regressions immediately).
Follow-up: How do you handle the attribution data at scale — it can be very large?

2. CLS Prevention — Dynamic Content

// React component that reserves space for async content
function AsyncWidget({ minHeight = 200 }) {
  const [data, setData] = useState(null);
  const [height, setHeight] = useState(minHeight);
  const ref = useRef();

  useEffect(() => {
    fetchData().then(setData);
  }, []);

  useLayoutEffect(() => {
    if (ref.current) {
      const actual = ref.current.getBoundingClientRect().height;
      if (actual > height) setHeight(actual);
    }
  }, [data]);

  return (
    <div style={{ minHeight: height, contain: 'layout' }} ref={ref}>
      {data ? <WidgetContent data={data} /> : <Skeleton height={minHeight} />}
    </div>
  );
}

// CSS containment prevents shifts from propagating
.widget-container {
  contain: layout style;  /* Shifts inside don't affect siblings */
  content-visibility: auto;  /* Skip rendering if off-screen */
  contain-intrinsic-size: 0 200px;  /* Estimated size when skipped */
}

3. INP Optimization — Yielding Pattern

// Pattern: Immediate feedback → yield → heavy work
async function handleFilterChange(filterValue) {
  // Step 1: Immediate visual feedback (fast)
  setActiveFilter(filterValue);
  showLoadingIndicator();

  // Step 2: Yield to main thread — browser paints the loading state
  await scheduler.yield?.() ?? new Promise(r => setTimeout(r, 0));

  // Step 3: Heavy computation (main thread free for other input in between)
  const filtered = filterLargeDataset(allData, filterValue);

  // Step 4: Update UI with results
  setFilteredData(filtered);
  hideLoadingIndicator();
}

// Using React 18's startTransition for INP
import { startTransition } from 'react';

function SearchInput() {
  const [query, setQuery] = useState('');
  const [results, setResults] = useState([]);

  function handleChange(e) {
    setQuery(e.target.value); // Urgent: update input immediately
    startTransition(() => {
      setResults(filterItems(e.target.value)); // Non-urgent: can be interrupted
    });
  }

  return <>
    <input value={query} onChange={handleChange} />
    <ResultList items={results} />
  </>;
}
