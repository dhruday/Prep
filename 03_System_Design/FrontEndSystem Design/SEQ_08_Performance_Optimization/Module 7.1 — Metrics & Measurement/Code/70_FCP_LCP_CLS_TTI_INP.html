Fix patterns for each Core Web Vital

// ─── Fix LCP ───
// 1. Preload hero image
<link rel="preload" as="image" href="/hero.webp" fetchpriority="high">
// 2. Add fetchpriority to img
<img src="/hero.webp" fetchpriority="high" loading="eager">

// ─── Fix CLS ───
// Reserve space for async content with aspect-ratio
.hero-image { aspect-ratio: 16/9; width: 100%; }  /* no shift when image loads */
// Reserve ad slot size explicitly
.ad-slot { min-height: 90px; }

// ─── Fix INP (long handler) ───
// Break up long synchronous work with scheduler.yield()
button.addEventListener('click', async () => {
  updateUIOptimistically();    // fast: sync
  await scheduler.yield();     // yield to browser for paint
  await heavyDataTransform(); // deferred heavy work
});

// ─── Measure all CWV in one call ───
import { onLCP, onINP, onCLS } from 'web-vitals';
[onLCP, onINP, onCLS].forEach(fn => fn(m =>
  navigator.sendBeacon('/cwv', JSON.stringify(m))
));
