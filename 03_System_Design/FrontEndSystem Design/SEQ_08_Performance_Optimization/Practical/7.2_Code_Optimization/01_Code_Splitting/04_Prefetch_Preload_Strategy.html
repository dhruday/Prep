Prefetch & Preload Strategy

Practical 7.2 · Resource hints for optimal loading priority

Resource Hint Comparison

HintPriorityWhenUse Case
<link rel="preload">High (mandatory)Current page, immediatelyCritical fonts, hero image, above-fold CSS
<link rel="prefetch">Low (idle time)Next navigation, idleNext page chunks, likely user paths
<link rel="preconnect">HighCurrent pageThird-party origins (CDN, API, fonts)
<link rel="dns-prefetch">LowBackgroundThird-party domains, fallback for preconnect
<link rel="modulepreload">HighCurrent pageES modules (parses + compiles too)

Correct Usage

<!-- HEAD: Resource hints -->

<!-- 1. Preconnect to critical origins (do DNS + TCP + TLS early) -->
<link rel="preconnect" href="https://fonts.googleapis.com">
<link rel="preconnect" href="https://api.example.com" crossorigin>

<!-- 2. Preload critical resources for THIS page -->
<link rel="preload" href="/fonts/inter.woff2" as="font" type="font/woff2" crossorigin>
<link rel="preload" href="/hero.webp" as="image">
<link rel="preload" href="/critical.css" as="style">

<!-- 3. Prefetch resources for NEXT likely page -->
<link rel="prefetch" href="/dashboard.chunk.js">
<link rel="prefetch" href="/api/user/dashboard" as="fetch">

<!-- ⚠️ Common mistakes: -->
<!-- ❌ Preloading too many resources (contends with critical path) -->
<!-- ❌ Missing 'as' attribute (browser can't prioritize correctly) -->
<!-- ❌ Missing 'crossorigin' for fonts (always CORS) -->
<!-- ❌ Prefetching on slow connections (check navigator.connection) -->

Programmatic Prefetch on Hover

// Prefetch route chunk when user hovers over a link
// Google uses this pattern (QuickLink library)

function PrefetchLink({ to, children }) {
  const prefetched = useRef(false);

  const handleHover = () => {
    if (prefetched.current) return;
    prefetched.current = true;

    // Check if connection is fast enough
    const conn = navigator.connection;
    if (conn?.saveData || conn?.effectiveType === '2g') return;

    // Create prefetch link
    const link = document.createElement('link');
    link.rel = 'prefetch';
    link.href = getChunkUrl(to);
    document.head.appendChild(link);
  };

  return (
    <Link to={to} onMouseEnter={handleHover} onFocus={handleHover}>
      {children}
    </Link>
  );
}

// Intersection Observer pattern (QuickLink-style):
// Prefetch any link that enters the viewport
const observer = new IntersectionObserver((entries) => {
  entries.filter(e => e.isIntersecting).forEach(entry => {
    const link = document.createElement('link');
    link.rel = 'prefetch';
    link.href = entry.target.href;
    document.head.appendChild(link);
    observer.unobserve(entry.target);
  });
});
document.querySelectorAll('a[href^="/"]').forEach(a => observer.observe(a));
