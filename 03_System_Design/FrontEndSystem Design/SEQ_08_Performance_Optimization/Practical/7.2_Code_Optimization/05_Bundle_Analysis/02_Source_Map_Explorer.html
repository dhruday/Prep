Setup & Usage

# Install
npm install --save-dev source-map-explorer

# Generate source maps in production build
# CRA: GENERATE_SOURCEMAP=true npm run build
# Vite: build.sourcemap = true in vite.config.ts

# Run analysis
npx source-map-explorer build/static/js/*.js
npx source-map-explorer build/static/js/*.js --html result.html
npx source-map-explorer build/static/js/*.js --json stats.json

# package.json script
{
  "scripts": {
    "analyze": "source-map-explorer 'build/static/js/*.js'",
    "analyze:json": "source-map-explorer 'build/static/js/*.js' --json | tee sme.json"
  }
}

# ⚠️ IMPORTANT: Strip source maps before deploying to production!
# Source maps expose your original source code
# Options:
# 1. Upload to error tracker (Sentry) then delete
# 2. Use hidden-source-map devtool in webpack
# 3. Restrict access via server config (same-origin only)
