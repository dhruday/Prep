react-window FixedSizeList

import { FixedSizeList as List } from 'react-window';

// Basic usage — renders only ~15-20 items regardless of list size
function VirtualList({ items }) {
  const Row = ({ index, style }) => (
    <div style={style}>
      {items[index].name} — ${items[index].price}
    </div>
  );

  return (
    <List
      height={400}          // viewport height
      width={"100%"}
      itemCount={items.length}
      itemSize={50}          // row height px (MUST be fixed)
      overscanCount={5}     // render 5 extra off-screen
    >
      {Row}
    </List>
  );
}

Why Virtualization Matters

// Without virtualization (10,000 items):
// - 10,000 DOM nodes created
// - ~500ms initial render on fast machine
// - ~2-5s on mobile devices
// - Scroll jank, memory pressure
//
// With react-window (10,000 items):
// - ~20 DOM nodes created (visible + overscan)
// - < 5ms initial render
// - Smooth 60fps scrolling
// - Constant memory usage
//
// Library comparison:
// react-window   — lightweight (6KB), simpler API
// react-virtuoso — auto height, grouped lists, more features
// @tanstack/virtual — headless, framework-agnostic
// react-virtualized — legacy (larger), use react-window instead
