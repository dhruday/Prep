react-window VariableSizeList

import { VariableSizeList as List } from 'react-window';

// You must provide itemSize function: index → pixel height
function ChatMessages({ messages }) {
  const listRef = useRef();

  // Pre-calculate or estimate heights
  const getItemSize = (index) => {
    const msg = messages[index];
    const lineCount = Math.ceil(msg.text.length / 60);
    return 40 + lineCount * 20; // base + lines × lineHeight
  };

  return (
    <List
      ref={listRef}
      height={500}
      width={"100%"}
      itemCount={messages.length}
      itemSize={getItemSize}     // function, not number
      overscanCount={3}
    >
      {({ index, style }) => (
        <div style={style}>
          <strong>{messages[index].user}</strong>
          <p>{messages[index].text}</p>
        </div>
      )}
    </List>
  );
}

Dynamic Height Measurement (react-virtuoso)

// react-virtuoso handles dynamic heights automatically
// No need to pre-calculate — it measures after render

import { Virtuoso } from 'react-virtuoso';

function Feed({ posts }) {
  return (
    <Virtuoso
      data={posts}
      itemContent={(index, post) => (
        <PostCard
          title={post.title}
          body={post.body}
          images={post.images} // images load → height changes → OK!
        />
      )}
      // Optional: provide estimate for better initial layout
      defaultItemHeight={200}
      // Optional: enable smooth scrolling to index
      initialTopMostItemIndex={posts.length - 1}
    />
  );
}
