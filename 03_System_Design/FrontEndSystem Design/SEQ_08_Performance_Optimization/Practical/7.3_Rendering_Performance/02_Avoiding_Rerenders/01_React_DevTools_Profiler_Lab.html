Programmatic Profiling with <Profiler>

import { Profiler } from 'react';

function onRender(id, phase, actualDur, baseDur, startTime, commitTime) {
  // phase: "mount" or "update"
  // actualDur: time spent rendering THIS update
  // baseDur: estimated render WITHOUT memoization
  
  if (actualDur > 16) { // > 1 frame (16ms)
    console.warn(`Slow render: ${id} took ${actualDur.toFixed(1)}ms`);
  }
  
  // Send to monitoring
  if (phase === 'update') {
    performance.mark(`react-${id}-start`);
    performance.measure(`React: ${id}`, { duration: actualDur });
  }
}

// Wrap the section you want to profile
<Profiler id="Dashboard" onRender={onRender}>
  <Dashboard />
</Profiler>

// ⚠️ Profiler has ~small overhead. Use in dev or selectively in prod.
// Google uses this to track render perf in YouTube, Search.
