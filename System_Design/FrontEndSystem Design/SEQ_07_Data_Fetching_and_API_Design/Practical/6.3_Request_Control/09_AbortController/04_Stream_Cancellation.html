📖 Reading a Stream with AbortController

const controller = new AbortController();

const res = await fetch(`${API}/api/streaming/ai-chat`, {
  signal: controller.signal
});

const reader = res.body
  .pipeThrough(new TextDecoderStream())
  .getReader();

while (true) {
  const { done, value } = await reader.read();
  if (done) break;
  processChunk(value);
}

// To cancel mid-stream:
controller.abort(); // reader.read() rejects with AbortError
Aborting the fetch signal causes the ReadableStream reader to reject. Always handle AbortError in your read loop.

🔧 Full SSE Stream + Cancel Implementation

let controller = null;

async function startStream() {
  controller = new AbortController();
  let bytes = 0, chunks = 0;

  try {
    const res = await fetch(url, { signal: controller.signal });
    const reader = res.body
      .pipeThrough(new TextDecoderStream())
      .getReader();

    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      chunks++;
      bytes += value.length;
      appendToOutput(value);
    }
  } catch (e) {
    if (e.name === 'AbortError') {
      log(`Cancelled after ${chunks} chunks (${bytes} bytes)`);
    }
  }
}

function stopStream() {
  if (controller) controller.abort();
}
