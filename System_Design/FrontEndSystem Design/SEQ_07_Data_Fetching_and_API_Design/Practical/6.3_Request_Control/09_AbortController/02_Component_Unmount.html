❌ Memory Leak — Updating Removed DOM

function mountComponent(container) {
  container.innerHTML = '<p>Loading...</p>';

  fetch(`${API}/api/search/slow?q=data`)
    .then(r => r.json())
    .then(data => {
      // ⚠️ Container may already be removed from DOM!
      container.innerHTML = `<p>${data.results.length} results</p>`;
      console.log('Updated DOM after unmount!'); // LEAK
    });
}

function unmountComponent(container) {
  container.innerHTML = ''; // Removed, but fetch still pending
}
The fetch callback fires after unmount, writing to detached DOM nodes. This leaks memory and can cause errors in frameworks.

✅ Clean Cleanup with AbortController

let controller = null;

function mountComponent(container) {
  controller = new AbortController();
  container.innerHTML = '<p>Loading...</p>';

  fetch(`${API}/api/search/slow?q=data`, {
    signal: controller.signal
  })
    .then(r => r.json())
    .then(data => {
      container.innerHTML = `<p>${data.results.length} results</p>`;
    })
    .catch(e => {
      if (e.name === 'AbortError') return; // Expected
      throw e;
    });
}

function unmountComponent(container) {
  if (controller) controller.abort(); // Cancel in-flight
  container.innerHTML = '';
}
The request is cancelled immediately on unmount. No callback fires, no leaked references, no wasted bandwidth.

📝 React useEffect Cleanup Pattern

function UserProfile({ userId }) {
  const [user, setUser] = useState(null);

  useEffect(() => {
    const controller = new AbortController();

    fetch(`/api/users/${userId}`, {
      signal: controller.signal
    })
      .then(r => r.json())
      .then(data => setUser(data))
      .catch(e => {
        if (e.name !== 'AbortError') console.error(e);
      });

    // Cleanup runs on unmount or dependency change
    return () => controller.abort();
  }, [userId]);

  return user ? `<div>${user.name}</div>` : 'Loading...';
}
Returning controller.abort() from useEffect ensures fetch is cancelled whenever the component unmounts or userId changes.
