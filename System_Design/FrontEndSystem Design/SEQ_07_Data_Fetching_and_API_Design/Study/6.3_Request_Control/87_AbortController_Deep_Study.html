1. AbortController / AbortSignal Architecture

// AbortController is a DOM API (not fetch-specific!)
// Pattern: Controller creates Signal, Signal drives cancellation

// ┌────────────────┐     ┌────────────────┐
// │ AbortController │────→│  AbortSignal   │
// │                │     │                │
// │ .abort()       │     │ .aborted       │ (boolean)
// │ .abort(reason) │     │ .reason        │ (any value)
// └────────────────┘     │ .throwIfAborted │ (throws if aborted)
//                        │ 'abort' event  │ (EventTarget)
//                        └────────────────┘

// The signal is passed to consumer APIs:
fetch(url, { signal })           // Network request
addEventListener(type, fn, { signal })  // Event listener
setTimeout(fn, ms, { signal })  // Timer (not yet standard)
stream.pipeTo(dest, { signal })  // Streams

// When controller.abort() is called:
// 1. signal.aborted becomes true
// 2. 'abort' event dispatched on signal
// 3. All consumers listening to signal react:
//    - fetch: TCP connection closed, Promise rejects with AbortError
//    - addEventListener: listener removed
//    - stream: stream cancelled

2. Network-Level Behavior

// What happens at the network level when fetch is aborted:

// 1. If request hasn't been sent yet (still in queue):
//    → Removed from queue, no network activity

// 2. If DNS lookup in progress:
//    → Lookup cancelled

// 3. If TCP connection being established:
//    → RST packet sent, connection torn down

// 4. If request sent, waiting for response:
//    → TCP connection closed (RST)
//    → Server may or may not process the request
//    → ⚠️ Non-idempotent requests may have side effects!

// 5. If response headers received, body streaming:
//    → Stop reading body, close connection
//    → Partial body data discarded

// Critical insight: abort doesn't guarantee server didn't process!
// For POST/PUT: server may have already executed the mutation
// Only safe for idempotent operations (GET) or with idempotency keys

3. AbortSignal.any() — Composite Signals

// New API: Combine multiple abort conditions

const userCancel = new AbortController();
const timeout = AbortSignal.timeout(5000);
const componentUnmount = new AbortController();

// Any of these signals aborting will cancel the fetch:
const combined = AbortSignal.any([
  userCancel.signal,
  timeout,
  componentUnmount.signal
]);

fetch(url, { signal: combined });

// Abort reasons are preserved:
// If timeout fires: signal.reason === DOMException('TimeoutError')
// If user cancels: signal.reason === controller.abort('user cancelled')

// Browser support (2024): Chrome 124+, Firefox 124+, Safari 17.4+
// Polyfill: manually listen to all signals and abort a combined controller

4. AbortSignal.timeout() — Static Factory

// Cleaner than manual setTimeout + AbortController:

// ❌ Old pattern:
const controller = new AbortController();
const timer = setTimeout(() => controller.abort(), 5000);
try {
  const res = await fetch(url, { signal: controller.signal });
} finally {
  clearTimeout(timer); // Must clean up timer!
}

// ✅ New pattern:
const res = await fetch(url, {
  signal: AbortSignal.timeout(5000)
});
// Automatically aborts after 5s
// Error: TimeoutError (different from AbortError)
// No cleanup needed — timer managed internally

// Combine with user cancel:
const res = await fetch(url, {
  signal: AbortSignal.any([
    user.signal,
    AbortSignal.timeout(5000)
  ])
});

5. Memory Leak Prevention

// AbortController + event listeners — cleanup pattern:

// ❌ Memory leak: listener persists after component unmount
useEffect(() => {
  window.addEventListener('resize', handleResize);
  return () => window.removeEventListener('resize', handleResize);
  // Problem: need same function reference for removal
});

// ✅ Clean: AbortController manages listener lifecycle
useEffect(() => {
  const controller = new AbortController();
  
  window.addEventListener('resize', handleResize, {
    signal: controller.signal // Auto-removed on abort
  });
  window.addEventListener('scroll', handleScroll, {
    signal: controller.signal // Same signal for multiple listeners
  });
  
  return () => controller.abort(); // One call removes ALL listeners
});
// No need to store function references!
// Works with anonymous functions too

1. Request Cancellation Hierarchy

// Multi-level cancellation architecture:

App Level ────────────────────────────────────────
  │ AbortController: logout, session timeout
  │
Route Level ──────────────────────────────────────
  │ AbortController: route navigation
  │ Cancel all requests from previous route
  │
Component Level ──────────────────────────────────
  │ AbortController: unmount, prop change
  │ useEffect cleanup
  │
Request Level ────────────────────────────────────
    AbortController: timeout, user cancel button

// Implementation with AbortSignal.any():
const appSignal = appController.signal;        // logout
const routeSignal = routeController.signal;    // navigation
const componentSignal = compController.signal;  // unmount
const timeoutSignal = AbortSignal.timeout(10000); // 10s

fetch(url, {
  signal: AbortSignal.any([
    appSignal, routeSignal, componentSignal, timeoutSignal
  ])
});

2. React Query + AbortSignal Integration

// React Query automatically passes AbortSignal to queryFn:

useQuery({
  queryKey: ['products', searchTerm],
  queryFn: ({ signal }) => {
    // signal is automatically aborted when:
    // 1. queryKey changes (searchTerm update)
    // 2. Component unmounts
    // 3. queryClient.cancelQueries() called
    // 4. Query becomes stale and new fetch starts
    
    return fetch(`/api/products?q=${searchTerm}`, { signal })
      .then(r => r.json());
  },
});

// For Axios:
queryFn: ({ signal }) => {
  return axios.get('/api/products', { signal });
  // Axios supports signal since v0.22
}

// Manual cancel from optimistic update:
await queryClient.cancelQueries({ queryKey: ['products'] });
// This aborts the signal passed to all matching queryFns

3. Search with Abort — Race Condition Prevention

// Complete search architecture with abort:

User types: "r" → "re" → "rea" → "reac" → "react"

Without abort:
  "r"     ──→ [200ms server] ──→ results arrive (STALE!)
  "re"    ──→ [150ms server] ──→ results arrive (STALE!)
  "rea"   ──→ [300ms server] ──→ results arrive (STALE!)
  "reac"  ──→ [100ms server] ──→ results arrive
  "react" ──→ [250ms server] ──→ results arrive ← CURRENT
  // "rea" returns AFTER "reac" → wrong results shown!

With abort:
  "r"     ──→ [ABORTED] ← canceled when "re" sent
  "re"    ──→ [ABORTED] ← canceled when "rea" sent
  "rea"   ──→ [ABORTED] ← canceled when "reac" sent
  "reac"  ──→ [ABORTED] ← canceled when "react" sent
  "react" ──→ [250ms server] ──→ results arrive ← ONLY request
  // Only the FINAL request completes
Q1: What is AbortController and how does it work at the browser level?
AbortController is a DOM API consisting of two parts:

AbortController: The control mechanism. Has one method: abort(reason?). Creates and owns an AbortSignal.
AbortSignal: The notification mechanism. An EventTarget with: .aborted (boolean), .reason (any), .throwIfAborted() method, and 'abort' event.

When controller.abort() is called, the signal's aborted flag is set to true, and an 'abort' event is dispatched. Any API consuming that signal can listen for the event and react (cancel network request, remove event listener, stop stream, etc.).

At the network level: fetch listens for the abort event. When fired, it sends a TCP RST to close the connection, stops reading the response body, and rejects the Promise with a DOMException named 'AbortError'.
Follow-up: Can you abort a request after the server has processed it?
Yes, but the server-side effect still happened. Abort only cancels the CLIENT-SIDE processing. If you abort() after the server received a POST request, the mutation may have been applied. This is why idempotency keys are critical for abortable mutations.
Q2: How do you handle AbortError vs other errors in a catch block?
Detection methods:
1. catch (err) { if (err.name === 'AbortError') } — DOMException named AbortError
2. catch (err) { if (signal.aborted) } — check signal state
3. For timeouts: err.name === 'TimeoutError' (from AbortSignal.timeout)

Best practice: AbortError should be SILENTLY ignored in most cases. It's an expected cancellation, not a real error.

Pattern:
try { const data = await fetch(url, { signal }); } catch (err) {
  if (err.name === 'AbortError') return; // Expected, ignore
  if (err.name === 'TimeoutError') showRetry(); // Timeout
  throw err; // Real error
}
Follow-up: What about AbortSignal.any() — which error do you get?
You get the reason from whichever signal aborted first. AbortSignal.timeout() produces TimeoutError, manual abort produces AbortError (or custom reason). Check signal.reason to distinguish which abort source triggered.
Q3: Design a search component that prevents race conditions using AbortController.
Architecture: debounce + abort + cache.

1. State: searchTerm (raw), debouncedTerm (after 300ms)
2. useEffect on debouncedTerm: create AbortController, fetch with signal
3. Cleanup: useEffect return → controller.abort() → cancels previous request
4. React Query: queryKey=[search, debouncedTerm], queryFn uses {signal} param
5. Cache: staleTime=5min → returning to previous search term uses cache
6. Min length: enabled: debouncedTerm.length >= 2

Why React Query is ideal: it handles abort automatically on queryKey change. When debouncedTerm changes from "rea" to "reac", existing "rea" query's signal is aborted, "reac" query starts fresh. No manual AbortController management needed.
Follow-up: What if you're NOT using React Query?
Manual pattern: store AbortController in useRef. In useEffect, abort previous ref, create new controller, fetch with signal. Return cleanup that aborts. Also store a request ID counter — on response, check if id matches latest to prevent state update from stale response.
Q4: How would you implement request timeouts with AbortController?
Three approaches:

1. AbortSignal.timeout(ms): const res = await fetch(url, { signal: AbortSignal.timeout(5000) }). Simplest, no cleanup. Throws TimeoutError.

2. Promise.race: Promise.race([fetch(url, {signal}), new Promise((_, reject) => setTimeout(() => { controller.abort(); reject(new Error('timeout')); }, 5000)))]). More control but needs cleanup.

3. AbortSignal.any + timeout: AbortSignal.any([userSignal, AbortSignal.timeout(5000)]). Best of both — supports user cancel AND timeout.

Production recommendation: AbortSignal.any with both user-controlled and timeout signals. Differentiate errors by checking signal.reason or err.name.
Q5: Beyond fetch, what other APIs support AbortSignal?
Browser APIs:
1. fetch() — cancel network requests
2. addEventListener(type, fn, {signal}) — auto-remove listeners
3. ReadableStream.pipeTo(writable, {signal}) — cancel streaming
4. EventSource — close SSE connection
5. navigator.locks.request(name, {signal}, cb) — cancel lock acquisition
6. Scheduler.postTask(fn, {signal}) — cancel scheduled tasks

Node.js APIs:
1. fs.readFile(path, {signal}) — cancel file reads
2. http.request(options, {signal}) — cancel HTTP requests
3. stream.pipeline(...args, {signal}) — cancel streams
4. child_process.exec(cmd, {signal}) — kill child processes

Libraries: Axios (.signal), tRPC (signal passthrough), Apollo Client (signal in context)
Follow-up: How does addEventListener + signal work?
When signal aborts, the listener is AUTOMATICALLY removed. No need for removeEventListener or matching function references. One controller.abort() can remove dozens of listeners added with the same signal. Perfect for useEffect cleanup.

1. useAbortableFetch Hook

function useAbortableFetch(url, options = {}) {
  const [data, setData] = useState(null);
  const [error, setError] = useState(null);
  const [loading, setLoading] = useState(true);
  
  useEffect(() => {
    const controller = new AbortController();
    
    setLoading(true);
    setError(null);
    
    fetch(url, { ...options, signal: controller.signal })
      .then(res => {
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        return res.json();
      })
      .then(json => {
        setData(json);
        setLoading(false);
      })
      .catch(err => {
        if (err.name === 'AbortError') return; // Silently ignore
        setError(err);
        setLoading(false);
      });
    
    return () => controller.abort();
    // ^ Cleanup: abort on unmount or url change
  }, [url]);
  
  return { data, error, loading };
}

2. Cascading Abort — Multiple Listeners

// One controller cleans up everything on unmount
useEffect(() => {
  const ctrl = new AbortController();
  const { signal } = ctrl;
  
  // Fetch data
  fetch('/api/data', { signal })
    .then(r => r.json())
    .then(setData)
    .catch(ignoreAbort);
  
  // Event listeners (auto-removed on abort)
  window.addEventListener('resize', handleResize, { signal });
  window.addEventListener('scroll', handleScroll, { signal });
  document.addEventListener('keydown', handleKey, { signal });
  
  // Polling interval
  const pollId = setInterval(() => {
    if (signal.aborted) return;
    fetch('/api/updates', { signal }).then(handleUpdate).catch(ignoreAbort);
  }, 5000);
  signal.addEventListener('abort', () => clearInterval(pollId));
  
  return () => ctrl.abort(); // ONE call cleans up EVERYTHING
}, []);

function ignoreAbort(err) {
  if (err.name !== 'AbortError') throw err;
}

3. Timeout + User Cancel Combined

async function fetchWithTimeout(url, { timeout = 10000, userSignal } = {}) {
  // Combine user cancel + timeout into one signal
  const signals = [AbortSignal.timeout(timeout)];
  if (userSignal) signals.push(userSignal);
  
  const combinedSignal = AbortSignal.any(signals);
  
  try {
    const response = await fetch(url, { signal: combinedSignal });
    if (!response.ok) throw new Error(`HTTP ${response.status}`);
    return await response.json();
  } catch (err) {
    if (err.name === 'TimeoutError') {
      throw new Error(`Request timed out after ${timeout}ms`);
    }
    if (err.name === 'AbortError') {
      throw new Error('Request cancelled by user');
    }
    throw err; // Network error or HTTP error
  }
}

// Usage:
const userCtrl = new AbortController();
cancelBtn.onclick = () => userCtrl.abort();
const data = await fetchWithTimeout('/api/slow', {
  timeout: 5000, userSignal: userCtrl.signal
});

Error Handling Pattern

try {
  const res = await fetch(url, { signal });
} catch (err) {
  if (err.name === 'AbortError')   → User/component cancelled (ignore)
  if (err.name === 'TimeoutError') → Timeout (show retry)
  if (err.name === 'TypeError')   → Network failure (offline)
  else                             → HTTP or parse error
}
