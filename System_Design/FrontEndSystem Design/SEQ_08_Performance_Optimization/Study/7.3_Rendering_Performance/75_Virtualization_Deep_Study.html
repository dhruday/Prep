1. The Large List Problem

// Rendering 10,000 items naively:
// - 10,000 DOM nodes created
// - Each node: ~0.5-2KB of memory (element + listeners + layout data)
// - Total: 5-20MB of DOM memory
// - Layout calculation: O(n) for any change
// - Painting: GPU must composite all visible + offscreen layers
// - React reconciliation: diffing 10,000 fiber nodes
//
// Real symptoms:
// - Initial render: 2-5 seconds (blocks main thread)
// - Scrolling: 10-30fps instead of 60fps (jank)
// - Filtering: 500ms+ delay to update list
// - Memory: tab uses 500MB+ → browser kills it on mobile
//
// Solution: VIRTUALIZATION
// Only render items currently visible in the viewport
// 10,000 items → only ~20-30 DOM nodes at any time
// As user scrolls, recycle DOM nodes with new data

2. How Virtualization Works

// Core concept: "windowing" — render only the visible window
//
// ┌─────────────────────────────────┐
// │  [invisible — above viewport]    │  Items 0-499 (NOT rendered)
// │  ...                             │
// ├─────────────────────────────────┤
// │  Item 500  ←── overscan top     │  Overscan buffer (5 items)
// │  Item 501                        │
// │  ══════════ VIEWPORT TOP ══════ │
// │  Item 505                        │  Visible items (20 items)
// │  Item 506                        │
// │  ...                             │
// │  Item 524                        │
// │  ══════════ VIEWPORT BOT ══════ │
// │  Item 525  ←── overscan bottom  │  Overscan buffer (5 items)
// │  Item 529                        │
// ├─────────────────────────────────┤
// │  [invisible — below viewport]    │  Items 530-9999 (NOT rendered)
// └─────────────────────────────────┘
//
// The container has a fixed height (viewport)
// An inner div has height = totalItems × itemHeight (creates scrollbar)
// Only ~30 items rendered, positioned with CSS transform/top
// On scroll: recalculate visible range, swap item data

react-window — Fixed Height List

import { FixedSizeList as List } from 'react-window';

const Row = React.memo(({ index, style }) => (
  <div style={style}>
    <ProductCard product={products[index]} />
  </div>
));

function ProductList({ products }) {
  return (
    <List
      height={600}        // Viewport height
      width={"100%"}
      itemCount={products.length}
      itemSize={80}        // Each row = 80px
      overscanCount={5}   // Render 5 extra items above/below
    >
      {Row}
    </List>
  );
}

@tanstack/react-virtual — Variable Height

import { useVirtualizer } from '@tanstack/react-virtual';

function DynamicList({ items }) {
  const parentRef = useRef(null);

  const virtualizer = useVirtualizer({
    count: items.length,
    getScrollElement: () => parentRef.current,
    estimateSize: () => 100,  // Estimated row height
    overscan: 5,
  });

  return (
    <div ref={parentRef} style={{ height: 600, overflow: 'auto' }}>
      <div style={{ height: virtualizer.getTotalSize(), position: 'relative' }}>
        {virtualizer.getVirtualItems().map(virtualRow => (
          <div
            key={virtualRow.key}
            ref={virtualizer.measureElement}  // Measures REAL height
            data-index={virtualRow.index}
            style={{
              position: 'absolute',
              top: 0,
              transform: `translateY(${virtualRow.start}px)`,
              width: '100%',
            }}
          >
            <ItemCard item={items[virtualRow.index]} />
          </div>
        ))}
      </div>
    </div>
  );
}

Infinite Scroll + Virtualization

// Combine virtualization with pagination
function InfiniteVirtualList() {
  const { data, fetchNextPage, hasNextPage, isFetching } =
    useInfiniteQuery({ queryKey: ['products'], queryFn: fetchProducts });

  const allItems = data?.pages.flatMap(p => p.items) ?? [];
  const parentRef = useRef(null);

  const virtualizer = useVirtualizer({
    count: hasNextPage ? allItems.length + 1 : allItems.length,
    getScrollElement: () => parentRef.current,
    estimateSize: () => 80,
    overscan: 10,
  });

  // Fetch more when user scrolls near the end
  useEffect(() => {
    const lastItem = virtualizer.getVirtualItems().at(-1);
    if (lastItem && lastItem.index >= allItems.length - 1 && hasNextPage && !isFetching) {
      fetchNextPage();
    }
  }, [virtualizer.getVirtualItems(), hasNextPage, isFetching]);

  return (
    <div ref={parentRef} style={{ height: 600, overflow: 'auto' }}>
      <div style={{ height: virtualizer.getTotalSize(), position: 'relative' }}>
        {virtualizer.getVirtualItems().map(vItem => (
          <div key={vItem.key} style={{ position: 'absolute', top: 0,
            transform: `translateY(${vItem.start}px)`, width: '100%' }}>
            {vItem.index < allItems.length
              ? <ProductCard product={allItems[vItem.index]} />
              : <LoadingSkeleton />
            }
          </div>
        ))}
      </div>
    </div>
  );
}

FAANG Interview Questions

Q1: How would you render a list of 100,000 items efficiently in React?
Virtualization — render only the items visible in the viewport (~20-30 items) plus a small overscan buffer. I'd use @tanstack/react-virtual: a headless virtualization library that's 5KB, supports variable heights, and works by calculating which items should be visible based on scroll position. The container has a fixed height, an inner div has the full computed height (creates proper scrollbar), and only visible items are rendered with absolute positioning. As the user scrolls, React recycles DOM nodes by swapping data. From 100K DOM nodes down to ~30 — memory drops from 200MB to ~2MB, initial render from 5s to 50ms, and scrolling stays at 60fps. Key optimization: React.memo on the row component to prevent re-rendering items that haven't changed data. For variable-height items, use the library's measurement API to dynamically calculate content size.
Follow-up: How would you handle search/filter with a virtualized list?
Q2: What are the tradeoffs of virtualization?
Several significant tradeoffs: (1) Search/find functionality breaks — Cmd+F only finds rendered DOM nodes, not virtualized-out items. Fix: implement custom search with scroll-to-index. (2) Accessibility challenges — screen readers may not detect all items. Fix: use ARIA attributes (aria-rowcount, aria-rowindex) to communicate total count. (3) Variable height complexity — if items have unknown height, you need a measurement pass (render → measure → position), which causes a layout flash. Use estimated sizes with progressive refinement. (4) Print support — printing only prints visible items. Fix: render all items on print with a print stylesheet. (5) Scroll position restoration — navigating away and back loses position. Fix: persist scroll offset in state/URL. (6) Dynamic content — items that change height after render (image load, accordion expand) need height recalculation callbacks. Despite these, for lists over 500 items, virtualization is essential for acceptable performance.
Follow-up: How does CSS contain property compare to full virtualization?

Virtual Grid (2D Virtualization)

// For data grids, image galleries, kanban boards
import { useVirtualizer } from '@tanstack/react-virtual';

function VirtualGrid({ items, columns = 4 }) {
  const parentRef = useRef(null);
  const rows = Math.ceil(items.length / columns);

  const rowVirtualizer = useVirtualizer({
    count: rows,
    getScrollElement: () => parentRef.current,
    estimateSize: () => 200,
    overscan: 3,
  });

  const columnVirtualizer = useVirtualizer({
    horizontal: true,
    count: columns,
    getScrollElement: () => parentRef.current,
    estimateSize: () => 250,
    overscan: 2,
  });

  return (
    <div ref={parentRef} style={{ height: 600, width: '100%', overflow: 'auto' }}>
      <div style={{ height: rowVirtualizer.getTotalSize(),
                    width: columnVirtualizer.getTotalSize(), position: 'relative' }}>
        {rowVirtualizer.getVirtualItems().map(vRow =>
          columnVirtualizer.getVirtualItems().map(vCol => {
            const idx = vRow.index * columns + vCol.index;
            if (idx >= items.length) return null;
            return (
              <div key={`${vRow.index}-${vCol.index}`} style={{
                position: 'absolute',
                top: 0, left: 0,
                transform: `translateX(${vCol.start}px) translateY(${vRow.start}px)`,
                width: vCol.size, height: vRow.size,
              }}>
                <ItemCard item={items[idx]} />
              </div>
            );
          })
        )}
      </div>
    </div>
  );
}

Virtualization Checklist

✅ Lists > 500 items use virtualization library
✅ @tanstack/react-virtual or react-window installed
✅ Row components wrapped in React.memo
✅ Overscan set to 3-5 items for smooth scrolling
✅ Variable heights use measurement API (not fixed estimates)
✅ Infinite scroll integrated with React Query pagination
✅ ARIA attributes for accessibility (aria-rowcount, aria-rowindex)
✅ Custom search replaces browser Cmd+F for virtualized lists
✅ Scroll position preserved on navigation
✅ Print stylesheet renders all items for printing
