Practical 7.2 · When and how to use React memoization hooks

Interactive Demo: Memoization vs No Memoization

useMemo: Memoize Expensive Computations

// ❌ Without useMemo: re-sorts on EVERY render
function ProductList({ products, sortBy }) {
  const sorted = products
    .slice()
    .sort((a, b) => a[sortBy] - b[sortBy]); // O(n log n) every render!
  return sorted.map(p => <Product key={p.id} {...p} />);
}

// ✅ With useMemo: only re-sorts when products or sortBy change
function ProductList({ products, sortBy }) {
  const sorted = useMemo(
    () => products.slice().sort((a, b) => a[sortBy] - b[sortBy]),
    [products, sortBy] // dependency array
  );
  return sorted.map(p => <Product key={p.id} {...p} />);
}

useCallback: Stable Function References

// useCallback = useMemo for functions
// useCallback(fn, deps) is identical to useMemo(() => fn, deps)

// ❌ Without: new function reference every render, breaks memo
function Parent() {
  const [count, setCount] = useState(0);
  const handleClick = () => console.log('clicked'); // new ref!
  return <MemoChild onClick={handleClick} />; // memo defeated!
}

// ✅ With: same function reference across renders
function Parent() {
  const [count, setCount] = useState(0);
  const handleClick = useCallback(
    () => console.log('clicked'),
    [] // no dependencies — stable forever
  );
  return <MemoChild onClick={handleClick} />; // memo works!
}

// When to use useCallback:
// ✅ Passing functions to React.memo'd children
// ✅ Functions in useEffect dependency arrays
// ✅ Functions passed to expensive hooks
// ❌ Every function (add overhead for no benefit)

When NOT to Memoize

// ❌ Cheap computation (< 1ms)
const fullName = useMemo(
  () => `${firstName} ${lastName}`,
  [firstName, lastName]
);
// String concat is instant — useMemo overhead > computation cost

// ❌ Values that change every render anyway
const now = useMemo(() => Date.now(), [trigger]);
// If trigger changes every render, memo never caches

// ❌ Simple state derivation (use derived state)
const isAdult = age >= 18; // Just compute inline

// Rule of thumb:
// "Don't useMemo unless profiling shows it's needed"
// — React team recommendation
//
// React Compiler (coming) will auto-memoize everything
