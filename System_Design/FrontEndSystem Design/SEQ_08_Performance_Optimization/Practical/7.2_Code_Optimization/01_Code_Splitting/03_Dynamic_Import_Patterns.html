Interactive: Dynamic Module Loading

Webpack Magic Comments

// Named chunks for debugging
const Chart = lazy(() => import(
  /* webpackChunkName: "chart-library" */
  './Chart'
));
// Output: chart-library.abc123.js instead of 42.abc123.js

// Prefetch: load in idle time (for likely next navigation)
const Settings = lazy(() => import(
  /* webpackPrefetch: true */
  './Settings'
));
// Adds: <link rel="prefetch" href="settings.js"> to <head>
// Loads AFTER current page resources, in idle time

// Preload: load in parallel with current chunk (high priority)
const Modal = lazy(() => import(
  /* webpackPreload: true */
  './Modal'
));
// Adds: <link rel="preload" href="modal.js">
// Loads IMMEDIATELY in parallel — use sparingly!

// Include mode: control chunk scope
const locale = await import(
  /* webpackInclude: /\/(en|fr|de)\.json$/ */
  /* webpackExclude: /\.test\./ */
  `./locales/${lang}.json`
);

// Eager mode: include in parent chunk (no separate file)
const mod = await import(
  /* webpackMode: "eager" */
  `./modules/${name}`
);
// No separate chunk — still async but bundled with parent

Vite Dynamic Imports

// Vite uses Rollup — different syntax for hints

// Named chunks via rollupOptions:
// vite.config.ts
export default defineConfig({
  build: {
    rollupOptions: {
      output: {
        manualChunks: {
          'vendor-react': ['react', 'react-dom'],
          'vendor-chart': ['chart.js', 'recharts'],
        }
      }
    }
  }
});

// Glob imports (Vite-specific):
const modules = import.meta.glob('./pages/*.tsx');
// Returns: { './pages/Home.tsx': () => import('./pages/Home.tsx'), ... }
// Each is lazy — loaded only when called
