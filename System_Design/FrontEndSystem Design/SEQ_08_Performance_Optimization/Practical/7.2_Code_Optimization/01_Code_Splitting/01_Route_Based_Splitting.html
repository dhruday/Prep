Route-Based Code Splitting

Practical 7.2 · Split bundles by route using React.lazy and dynamic imports

React Router + React.lazy

// Before: all routes in one bundle
import Home from './pages/Home';
import Dashboard from './pages/Dashboard';      // 200KB
import AdminPanel from './pages/AdminPanel';    // 150KB
import Analytics from './pages/Analytics';      // 300KB (chart lib)
// Total: 700KB loaded on first visit, even if user only sees Home

// After: split by route
import { lazy, Suspense } from 'react';

const Home = lazy(() => import('./pages/Home'));
const Dashboard = lazy(() => import('./pages/Dashboard'));
const AdminPanel = lazy(() => import('./pages/AdminPanel'));
const Analytics = lazy(() => import('./pages/Analytics'));

function App() {
  return (
    <Suspense fallback={<LoadingSpinner />}>
      <Routes>
        <Route path="/" element={<Home />} />
        <Route path="/dashboard" element={<Dashboard />} />
        <Route path="/admin" element={<AdminPanel />} />
        <Route path="/analytics" element={<Analytics />} />
      </Routes>
    </Suspense>
  );
}

// Result: Home page loads ~50KB
// Dashboard chunk loads only when navigating to /dashboard
// Analytics chunk (300KB) only loads when user visits /analytics

Next.js Automatic Route Splitting

// Next.js automatically splits by page/route
// app/
//   page.tsx         → / route chunk
//   dashboard/
//     page.tsx       → /dashboard chunk
//   admin/
//     page.tsx       → /admin chunk
//
// Each page.tsx becomes its own JS chunk automatically
// Shared dependencies go into a common chunk
// No manual React.lazy needed for routes!
//
// For component-level splitting within a page:
import dynamic from 'next/dynamic';

const HeavyChart = dynamic(() => import('../components/HeavyChart'), {
  loading: () => <p>Loading chart...</p>,
  ssr: false, // Don't render server-side (chart needs DOM)
});

Angular Lazy Routes

// Angular: lazy-load feature modules per route
const routes: Routes = [
  { path: '', component: HomeComponent },
  {
    path: 'dashboard',
    loadChildren: () =>
      import('./dashboard/dashboard.module')
        .then(m => m.DashboardModule)
  },
  {
    path: 'admin',
    loadComponent: () =>  // Angular 14+ standalone
      import('./admin/admin.component')
        .then(c => c.AdminComponent)
  },
];
