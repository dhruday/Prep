Component-Based Code Splitting

Practical 7.2 · Split heavy components that aren't needed on initial render

Pattern 1: Split on User Action (Modal)

// Heavy modal with rich text editor (200KB)
// Only 5% of users actually open it

const RichEditor = lazy(() => import('./RichEditor'));

function CommentSection() {
  const [showEditor, setShowEditor] = useState(false);

  return (
    <div>
      <button onClick={() => setShowEditor(true)}>
        Write Comment
      </button>

      {showEditor && (
        <Suspense fallback={<EditorSkeleton />}>
          <RichEditor />  {/* Loaded only when button clicked */}
        </Suspense>
      )}
    </div>
  );
}

Pattern 2: Split Below the Fold

// Components not visible on initial viewport
const Reviews = lazy(() => import('./Reviews'));
const RelatedProducts = lazy(() => import('./RelatedProducts'));
const Footer = lazy(() => import('./Footer'));

function ProductPage() {
  return (
    <div>
      {/* Above fold — loaded immediately */}
      <ProductHero />
      <PriceCard />
      <AddToCartButton />

      {/* Below fold — lazy loaded */}
      <Suspense fallback={<SectionSkeleton />}>
        <Reviews />
        <RelatedProducts />
        <Footer />
      </Suspense>
    </div>
  );
}

Pattern 3: Split by Feature Flag / Role

// Admin tools (150KB) only loaded for admin users
const AdminTools = lazy(() => import('./AdminTools'));

function Dashboard({ user }) {
  return (
    <div>
      <DashboardContent />
      {user.role === 'admin' && (
        <Suspense fallback={<div>Loading admin tools...</div>}>
          <AdminTools />
        </Suspense>
      )}
    </div>
  );
}

// Pattern 4: Intersection Observer trigger
function LazySection({ importFn, fallback }) {
  const [Component, setComponent] = useState(null);
  const ref = useRef(null);

  useEffect(() => {
    const observer = new IntersectionObserver(([entry]) => {
      if (entry.isIntersecting) {
        importFn().then(mod => setComponent(() => mod.default));
        observer.disconnect();
      }
    }, { rootMargin: '200px' }); // Load 200px before visible
    observer.observe(ref.current);
    return () => observer.disconnect();
  }, []);

  return <div ref={ref}>{Component ? <Component /> : fallback}</div>;
}
