Lazy loading — defer what isn't needed yet

Lazy loading delays loading a resource (component, image, module) until it's actually needed. This reduces the initial payload, speeds up FCP/LCP, and saves bandwidth for resources the user may never access.

Lazy loading patterns

// ─── React: lazy component with Suspense ───
const ChartDashboard = React.lazy(() =>
  import('./ChartDashboard') // chunk loaded only when rendered
);
{showChart && (
  <Suspense fallback={<ChartSkeleton />}>
    <ChartDashboard data={data} />
  </Suspense>
)}

// ─── Angular: lazy route ───
const routes: Routes = [
  {
    path: 'admin',
    loadComponent: () =>
      import('./admin/admin.component').then(m => m.AdminComponent),
  },
  {
    path: 'reports',
    loadChildren: () => import('./reports/routes'), // lazy module
  },
];

// ─── Custom IntersectionObserver lazy load ───
const observer = new IntersectionObserver((entries) => {
  entries.forEach(entry => {
    if (entry.isIntersecting) {
      const img = entry.target as HTMLImageElement;
      img.src = img.dataset.src!;   // swap placeholder → real src
      observer.unobserve(img);
    }
  });
}, { rootMargin: '200px' });  // pre-load 200px before visible

document.querySelectorAll('img[data-src]').forEach(img => observer.observe(img));
