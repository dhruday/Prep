# SHA-384 (recommended):
cat script.js | openssl dgst -sha384 -binary | openssl base64 -A
# Output: vtXRMe3mGCbOkNs0S4pZwRLNUfTp...

# Full integrity attribute:
echo -n "sha384-$(cat script.js | openssl dgst -sha384 -binary | openssl base64 -A)"

// ═══ Method 2: shasum (CLI) ═══
shasum -b -a 384 script.js | xxd -r -p | base64

// ═══ Method 3: Node.js ═══
const crypto = require('crypto');
const fs = require('fs');
const content = fs.readFileSync('script.js');
const hash = crypto.createHash('sha384').update(content).digest('base64');
console.log(`sha384-\${hash}`);

// ═══ Method 4: Web Crypto API (browser) ═══
async function computeSRI(text, algo = 'SHA-384') {
  const bytes = new TextEncoder().encode(text);
  const hash = await crypto.subtle.digest(algo, bytes);
  const b64 = btoa(String.fromCharCode(...new Uint8Array(hash)));
  return `\${algo.toLowerCase().replace('-','')}-\${b64}`;
}

Build Integration Script

// generate-sri.js — Node.js script for CI/CD
const crypto = require('crypto');
const fs = require('fs');
const path = require('path');

function generateSRI(filePath, algorithm = 'sha384') {
  const content = fs.readFileSync(filePath);
  const hash = crypto.createHash(algorithm).update(content).digest('base64');
  return `\${algorithm}-\${hash}`;
}

// Process all JS/CSS files in dist/
const distDir = path.join(__dirname, 'dist');
const manifest = {};

fs.readdirSync(distDir)
  .filter(f => /\.(js|css)$/.test(f))
  .forEach(file => {
    manifest[file] = generateSRI(path.join(distDir, file));
  });

// Write SRI manifest
fs.writeFileSync(
  path.join(distDir, 'sri-manifest.json'),
  JSON.stringify(manifest, null, 2)
);

console.log('SRI manifest generated:', manifest);

Q: Why SHA-384 and not SHA-256 for SRI?

SHA-384 is the recommended algorithm for SRI. Why not SHA-256? (1) Security margin: SHA-384 is a truncated SHA-512, running on 64-bit words. On 64-bit CPUs (all modern machines), SHA-512/384 is actually faster than SHA-256. More security for free performance. (2) Length extension attacks: SHA-256 is vulnerable to length extension attacks. SHA-384 (being truncated) is not. While not directly exploitable in SRI context, it's defense-in-depth. (3) Collision resistance: SHA-384 provides 192-bit collision resistance vs SHA-256's 128-bit. For a security-critical feature like SRI, extra margin matters. (4) Why not SHA-512? SHA-384 is slightly shorter hash string (less HTML bloat), while providing more than sufficient security. Diminishing returns. (5) Browser support: All major browsers support SHA-256, SHA-384, and SHA-512 for SRI. SHA-384 balances security and practicality.
