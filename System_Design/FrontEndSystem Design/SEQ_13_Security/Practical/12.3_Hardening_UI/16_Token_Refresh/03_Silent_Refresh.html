// ═══ Technique 2: Refresh token endpoint ═══
// Simpler and more reliable
async function silentRefresh() {
  const resp = await fetch('/api/auth/refresh', {
    method: 'POST',
    credentials: 'same-origin', // sends HttpOnly cookie
  });
  if (resp.ok) {
    const { accessToken } = await resp.json();
    setAccessToken(accessToken);
  } else {
    // Refresh failed → redirect to login
    redirectToLogin();
  }
}

Proactive vs Reactive Refresh

// REACTIVE: Wait for 401, then refresh
// ❌ Problem: User sees brief error/loading state

// PROACTIVE: Refresh BEFORE token expires
// ✅ Seamless experience — user never sees 401

// Strategy: Refresh at 75% of token lifetime
function scheduleProactiveRefresh(token) {
  const payload = JSON.parse(atob(token.split('.')[1]));
  const exp = payload.exp * 1000; // to milliseconds
  const now = Date.now();
  const lifetime = exp - now;
  const refreshAt = lifetime * 0.75; // 75% through lifetime

  setTimeout(async () => {
    await silentRefresh();
  }, refreshAt);
}

// Sliding window: Reset timer on user activity
let idleTimer;
function resetIdleTimer() {
  clearTimeout(idleTimer);
  idleTimer = setTimeout(() => {
    // User idle for 30 min → log out
    logout();
  }, 30 * 60 * 1000);
}
document.addEventListener('mousemove', resetIdleTimer);
document.addEventListener('keypress', resetIdleTimer);

Complete Token Manager

class TokenManager {
  #accessToken = null;
  #refreshTimer = null;
  #refreshPromise = null; // prevent concurrent refreshes

  async getAccessToken() {
    if (this.#isExpired()) {
      await this.refresh();
    }
    return this.#accessToken;
  }

  async refresh() {
    // Deduplicate: if refresh is already in progress, reuse it
    if (this.#refreshPromise) return this.#refreshPromise;

    this.#refreshPromise = this.#doRefresh();
    try {
      return await this.#refreshPromise;
    } finally {
      this.#refreshPromise = null;
    }
  }

  async #doRefresh() {
    const resp = await fetch('/api/auth/refresh', {
      method: 'POST',
      credentials: 'same-origin',
    });
    if (!resp.ok) throw new Error('Refresh failed');
    const { accessToken } = await resp.json();
    this.#setToken(accessToken);
  }

  #setToken(token) {
    this.#accessToken = token;
    this.#scheduleRefresh(token);
  }

  #scheduleRefresh(token) {
    clearTimeout(this.#refreshTimer);
    const { exp } = JSON.parse(atob(token.split('.')[1]));
    const ms = (exp * 1000 - Date.now()) * 0.75;
    this.#refreshTimer = setTimeout(() => this.refresh(), ms);
  }
}

Google Senior Frontend — Silent Refresh

Q: How do you handle token refresh in a React app with concurrent requests?

Problem: 5 API requests fire simultaneously. All get 401s. Without deduplication: 5 refresh calls, race conditions, wasted tokens. Solution: Request queue pattern. (1) First 401 triggers refresh. (2) Subsequent 401s are queued (Promise chaining — they await the same refresh Promise). (3) Refresh completes → all queued requests retry with new token. Implementation: Axios interceptor or custom fetch wrapper. refreshPromise = null. On 401: if (!refreshPromise) refreshPromise = doRefresh(). await refreshPromise. retry(). In finally: refreshPromise = null. This ensures exactly ONE refresh per expiration, regardless of concurrent requests. Auth0 SDK, Firebase Auth, and most production auth libraries implement this pattern internally.
