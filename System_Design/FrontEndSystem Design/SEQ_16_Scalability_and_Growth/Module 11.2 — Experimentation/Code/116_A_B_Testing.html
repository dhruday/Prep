A/B testing: measure impact of changes on real users before committing

A/B testing splits users into groups (A=control, B=variant) and measures a key metric (conversion, engagement, error rate). The split must be deterministic — the same user always sees the same variant. Split should be based on a stable identifier like userId, not sessionId. Statistical significance (p < 0.05) is required before declaring a winner.

Deterministic bucket assignment + event tracking

// ─── Hash-based deterministic assignment ───
function getVariant(userId: string, experimentId: string): 'A' | 'B' {
  let hash = 0;
  const key = `${userId}:${experimentId}`;
  for (const ch of key) hash = (hash * 31 + ch.charCodeAt(0)) & 0xffffffff;
  return (hash & 0xff) < 128 ? 'A' : 'B'; // 50/50 split
}

// ─── React hook ───
function useExperiment(id: string) {
  const userId = useUserId();
  const variant = getVariant(userId, id);

  useEffect(() => {
    analytics.track('experiment_assigned', { id, variant }); // exposure event
  }, [id, variant]);

  return variant;
}

// ─── Usage ───
function CheckoutButton() {
  const variant = useExperiment('checkout_cta_v2');
  const handleClick = () => analytics.track('checkout_started', { variant });
  return <button onClick={handleClick}>
    {variant === 'B' ? 'Buy Now ⚡' : 'Proceed to Checkout'}
  </button>;
}
