Edge rendering: generate personalized HTML at the CDN edge — no origin round-trip, no JS hydration needed

Traditional SSR hits your origin server on every request — adds 100–400ms latency. Static (SSG/ISR) can't personalize. Edge rendering combines the best of both: Workers (Cloudflare, Vercel Edge, Deno Deploy) run V8 isolates at 200+ PoPs worldwide. They personalize HTML based on cookies/headers/geolocation before the response ever touches a browser — no cold start, no DB lock-in, typically <10ms compute. The result is the TTFB of a CDN edge with the personalization of SSR.

Cloudflare Worker edge rendering + Next.js middleware

// ─── Cloudflare Worker — A/B at edge ───
export default {
  async fetch(req: Request, env: Env) {
    const url = new URL(req.url);
    const cookie = req.headers.get('Cookie') ?? '';
    let variant = parseCookie(cookie, 'ab_variant');

    if (!variant) {
      variant = Math.random() < 0.5 ? 'A' : 'B';
    }
    url.pathname = variant === 'B'
      ? '/lp-variant-b' : url.pathname; // fetch different static page

    const res = await fetch(url.toString());
    const newRes = new Response(res.body, res);
    newRes.headers.append('Set-Cookie', `ab_variant=${variant}; Path=/; Max-Age=86400`);
    return newRes;
  }
};

// ─── Next.js middleware (runs at Vercel Edge) ───
export function middleware(req: NextRequest) {
  const locale = req.headers.get('Accept-Language')?.slice(0, 2) ?? 'en';
  if (!req.nextUrl.pathname.startsWith(`/${locale}`)) {
    return NextResponse.redirect(new URL(`/${locale}${req.nextUrl.pathname}`, req.url));
  }
}
