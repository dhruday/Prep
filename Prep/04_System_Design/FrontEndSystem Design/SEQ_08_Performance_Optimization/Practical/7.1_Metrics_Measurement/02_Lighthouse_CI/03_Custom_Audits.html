Custom Lighthouse Audits

Practical 7.1 · Create custom audits for organization-specific perf checks

1. Custom Audit: Maximum Third-Party Scripts

// third-party-audit.js
// Fails if page loads more than 3 third-party scripts

import { Audit } from 'lighthouse';

class ThirdPartyScriptAudit extends Audit {
  static get meta() {
    return {
      id: 'third-party-script-count',
      title: 'Page loads limited third-party scripts',
      failureTitle: 'Too many third-party scripts loaded',
      description: 'Each 3rd-party script adds latency and risk.',
      requiredArtifacts: ['devtoolsLogs'],
    };
  }

  static async audit(artifacts) {
    const devtoolsLog = artifacts.devtoolsLogs[Audit.DEFAULT_PASS];
    const networkRecords = await NetworkRecords.request(devtoolsLog);

    const pageHost = new URL(artifacts.URL.finalUrl).host;
    const thirdParty = networkRecords.filter(req =>
      req.resourceType === 'Script' &&
      new URL(req.url).host !== pageHost
    );

    const MAX = 3;
    return {
      score: thirdParty.length <= MAX ? 1 : 0,
      numericValue: thirdParty.length,
      displayValue: `${thirdParty.length} third-party scripts`,
      details: Audit.makeTableDetails(
        [{ key: 'url', label: 'Script URL' },
         { key: 'size', label: 'Size (KB)' }],
        thirdParty.map(r => ({
          url: r.url,
          size: Math.round(r.transferSize / 1024),
        }))
      ),
    };
  }
}

2. Custom Audit: No Render-Blocking CSS

// render-blocking-css-audit.js
class RenderBlockingCSSAudit extends Audit {
  static get meta() {
    return {
      id: 'no-render-blocking-css',
      title: 'No render-blocking stylesheets above 5KB',
      failureTitle: 'Large render-blocking stylesheets found',
      description: 'Inline critical CSS and async-load the rest.',
      requiredArtifacts: ['TagsBlockingFirstPaint'],
    };
  }

  static audit(artifacts) {
    const blocking = artifacts.TagsBlockingFirstPaint
      .filter(tag => tag.tag.endsWith('link') && tag.transferSize > 5120);

    return {
      score: blocking.length === 0 ? 1 : 0,
      displayValue: `${blocking.length} blocking stylesheets > 5KB`,
    };
  }
}

3. Registering Custom Audits

// custom-config.js — Lighthouse custom config
module.exports = {
  extends: 'lighthouse:default',
  audits: [
    './third-party-audit.js',
    './render-blocking-css-audit.js',
  ],
  categories: {
    'custom-perf': {
      title: 'Custom Performance Checks',
      auditRefs: [
        { id: 'third-party-script-count', weight: 1 },
        { id: 'no-render-blocking-css', weight: 1 },
      ],
    },
  },
};

// Run: lighthouse https://example.com --config-path=./custom-config.js

4. Lighthouse Plugin Structure

// For reusable audits, package as a Lighthouse plugin:
//
// lighthouse-plugin-my-company/
// ├── package.json          ← name: "lighthouse-plugin-my-company"
// ├── plugin.js             ← exports { audits, category }
// └── audits/
//     ├── third-party.js
//     └── render-blocking.js
//
// plugin.js:
module.exports = {
  audits: [
    { path: 'lighthouse-plugin-my-company/audits/third-party.js' },
    { path: 'lighthouse-plugin-my-company/audits/render-blocking.js' },
  ],
  category: {
    title: 'My Company Perf',
    auditRefs: [
      { id: 'third-party-script-count', weight: 1 },
      { id: 'no-render-blocking-css', weight: 1 },
    ],
  },
};

// Run: lighthouse https://example.com --plugins=lighthouse-plugin-my-company
