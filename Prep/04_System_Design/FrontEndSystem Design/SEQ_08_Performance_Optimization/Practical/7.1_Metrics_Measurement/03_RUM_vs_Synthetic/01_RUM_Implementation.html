// Minimal RUM collector — drop into any page

(function() {
  const metrics = {};

  // 1. Navigation Timing
  window.addEventListener('load', () => {
    const nav = performance.getEntriesByType('navigation')[0];
    metrics.dns = nav.domainLookupEnd - nav.domainLookupStart;
    metrics.tcp = nav.connectEnd - nav.connectStart;
    metrics.ttfb = nav.responseStart - nav.requestStart;
    metrics.domLoad = nav.domContentLoadedEventEnd - nav.startTime;
    metrics.fullLoad = nav.loadEventEnd - nav.startTime;
  });

  // 2. Core Web Vitals via PerformanceObserver
  function observe(type, cb) {
    try {
      new PerformanceObserver((list) => {
        const entries = list.getEntries();
        cb(entries[entries.length - 1]);
      }).observe({ type, buffered: true });
    } catch(e) {}
  }

  observe('largest-contentful-paint', (e) => {
    metrics.lcp = Math.round(e.startTime);
  });

  let cls = 0;
  observe('layout-shift', (e) => {
    if (!e.hadRecentInput) cls += e.value;
    metrics.cls = Math.round(cls * 1000) / 1000;
  });

  let inp = 0;
  try {
    new PerformanceObserver((list) => {
      for (const e of list.getEntries()) {
        if (e.duration > inp) inp = e.duration;
      }
      metrics.inp = inp;
    }).observe({ type: 'event', durationThreshold: 16, buffered: true });
  } catch(e) {}

  // 3. Beacon on page unload
  document.addEventListener('visibilitychange', () => {
    if (document.visibilityState === 'hidden') {
      const payload = JSON.stringify({
        url: location.href,
        timestamp: Date.now(),
        metrics,
        connection: navigator.connection?.effectiveType,
        deviceMemory: navigator.deviceMemory,
      });
      navigator.sendBeacon('/api/rum', payload);
      // sendBeacon survives page unload
    }
  });
})();
