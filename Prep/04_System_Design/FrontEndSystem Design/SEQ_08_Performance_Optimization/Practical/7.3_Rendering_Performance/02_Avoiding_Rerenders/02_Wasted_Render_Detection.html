Top 5 Wasted Render Patterns

// Pattern 1: State too high in tree
// ❌ Search state in App → entire page re-renders
function App() {
  const [search, setSearch] = useState(''); // Lifting too high!
  return (
    <>
      <Header /> // re-renders on every keystroke!
      <SearchBox value={search} onChange={setSearch} />
      <ProductGrid /> // re-renders on every keystroke!
      <Footer /> // re-renders on every keystroke!
    </>
  );
}

// ✅ Fix: Colocate state with the component that uses it
function SearchSection() {
  const [search, setSearch] = useState(''); // local!
  return <SearchBox value={search} onChange={setSearch} />;
}

// Pattern 2: Inline objects / arrays as props
// ❌ New object every render
<Chart config={{ color: 'red', size: 100 }} />

// ✅ Hoist to module scope or useMemo
const CHART_CONFIG = { color: 'red', size: 100 };
<Chart config={CHART_CONFIG} />

// Pattern 3: Context providing object values
// ❌ New object → all consumers re-render
<Ctx.Provider value={{ user, theme }}>

// ✅ Memoize or split contexts
const value = useMemo(() => ({ user, theme }), [user, theme]);
<Ctx.Provider value={value}>

why-did-you-render Library

// Install: npm install @welldone-software/why-did-you-render

// wdyr.js (import BEFORE React in dev only)
import React from 'react';

if (process.env.NODE_ENV === 'development') {
  const whyDidYouRender = require('@welldone-software/why-did-you-render');
  whyDidYouRender(React, {
    trackAllPureComponents: true,  // track all React.memo components
    logOnDifferentValues: true,
  });
}

// Per-component tracking:
ProductList.whyDidYouRender = true;

// ProductList

//     (though values are equal)
