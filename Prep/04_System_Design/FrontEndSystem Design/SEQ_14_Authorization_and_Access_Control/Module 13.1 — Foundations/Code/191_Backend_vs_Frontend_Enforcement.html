The enforcement rule: Frontend = UX, Backend = Security

🖥️ Frontend enforcement — UX ONLY
✅ Hides forbidden buttons
✅ Blocks route navigation
✅ Shows "Access Denied" message
✅ Disables form fields for read-only
❌ NOT security — bypassable with DevTools or direct API calls
🔒 Backend enforcement — SECURITY
✅ Validates token (AuthN)
✅ Checks role for endpoint (AuthZ)
✅ Verifies resource ownership
✅ Applies row-level security
✅ Strips unauthorized fields from response
✅ Works regardless of what UI shows
⚠️ Key insight: Design backend as if frontend doesn't exist
If you deleted the frontend entirely, every API endpoint should still correctly return 403 for unauthorized requests. The backend must be the source of truth — not a validator that trusts frontend state.

Multi-layer backend authorization

// Express.js — authentication + route + resource ownership chain
function requirePermission(perm: string) {
  return (req: Request, res: Response, next: NextFunction) => {
    if (!req.user.permissions.includes(perm))
      return res.status(403).json({ error: 'Forbidden' });
    next();
  };
}

async function requireDocOwnership(req, res, next) {
  const doc = await db.documents.findById(req.params.id);
  if (!doc) return res.status(404).json({ error: 'Not found' });
  const ok = doc.ownerId === req.user.id
           || doc.collaborators.includes(req.user.id)
           || req.user.permissions.includes('document:read:all');
  if (!ok) return res.status(403).json({ error: 'Forbidden' });
  req.document = doc; next();
}

// Route: AuthN → Route AuthZ → Resource AuthZ
router.delete('/docs/:id',
  authenticate,                      // is token valid?
  requirePermission('document:delete'), // has role?
  requireDocOwnership,               // owns this doc? ← prevents IDOR
  (req, res) => res.status(204).send()
);
