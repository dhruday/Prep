Key difference
MVC: Controller mediates. View pulls from Model via Controller.

MVVM: ViewModel exposes observable state. View binds declaratively. Two-way data binding — Angular: [(ngModel)]

Angular MVVM Pattern (TypeScript)

// MODEL — pure data/service logic
@Injectable({ providedIn: 'root' })
export class UserService {
  private users$ = new BehaviorSubject<User[]>([]);
  
  loadUsers(): Observable<User[]> {
    return this.http.get<User[]>('/api/users').pipe(
      tap(users => this.users$.next(users))
    );
  }
}

// VIEWMODEL — Angular component class transforms data for view
@Component({ selector: 'app-user-list', templateUrl: './user-list.component.html' })
export class UserListComponent {
  users$ = this.userService.users$.pipe(
    map(users => users.filter(u => u.active))  // transform for view
  );
  searchTerm = '';
  
  constructor(private userService: UserService) {}
}

// VIEW — HTML template, declaratively bound to ViewModel
// user-list.component.html:
// <input [(ngModel)]="searchTerm">
// <user-card *ngFor="let u of users$ | async" [user]="u"></user-card>
