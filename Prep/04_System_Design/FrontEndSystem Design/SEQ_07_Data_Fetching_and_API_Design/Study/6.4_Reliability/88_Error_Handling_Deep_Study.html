1. Error Classification — The Foundation

// First principle: NOT all errors should be retried
// Classify before acting — this is what separates senior from junior

// TRANSIENT errors — retry makes sense
// ├─ NETWORK_TIMEOUT  → might succeed next time
// ├─ SERVER_UNAVAILABLE (503) → service restarting
// ├─ RATE_LIMITED (429) → retry after backoff
// └─ NETWORK_ERROR  → DNS failure, connectivity blip

// PERMANENT errors — never retry, show actionable message
// ├─ NOT_FOUND (404) → resource doesn't exist
// ├─ FORBIDDEN (403) → user lacks permissions
// ├─ UNAUTHORIZED (401) → session expired → redirect to login
// ├─ VALIDATION_ERROR (400) → bad request from client
// └─ CONFLICT (409) → business rule conflict

// UNKNOWN — log and treat as permanent
function classifyError(error) {
  if (error instanceof TypeError && error.message === 'Failed to fetch') {
    return 'NETWORK_ERROR';  // Browser couldn't connect at all
  }
  if (error instanceof DOMException && error.name === 'AbortError') {
    return 'CANCELLED';      // Don't retry user cancellations
  }
  if (error.status) {
    const map = {
      400: 'VALIDATION', 401: 'UNAUTHORIZED',
      403: 'FORBIDDEN',  404: 'NOT_FOUND',
      409: 'CONFLICT',   429: 'RATE_LIMITED',
      503: 'SERVER_UNAVAILABLE',
    };
    return map[error.status] ?? (error.status >= 500 ? 'SERVER_ERROR' : 'UNKNOWN');
  }
  return 'UNKNOWN';
}

2. Exponential Backoff with Jitter

// WHY exponential? Linear retry (1s, 1s, 1s) creates synchronized storms
// WHY jitter? Even exponential (1s, 2s, 4s) syncs clients if they all started together
// Full jitter: random value in [0, exponentialDelay] de-syncs all clients

// Timeline WITHOUT jitter (1000 clients retrying server that was down 5s):
//   t=5s:  [1000 clients retry simultaneously] → server crushed again
//   t=7s:  [1000 clients retry simultaneously] → still down
//   t=11s: [1000 clients retry simultaneously] → thundering herd

// Timeline WITH full jitter (1000 clients):
//   t=5s:  [~100 clients retry at random intervals across 0-1s]
//   t=7s:  [~100 clients spread across 0-2s window]
//   t=11s: [~100 clients spread across 0-4s window]
//   Server recovers incrementally, load ramps up smoothly
function calculateBackoff(attempt, { baseDelay = 1000, maxDelay = 30000, jitter = true } = {}) {
  const exponential = Math.min(baseDelay * Math.pow(2, attempt), maxDelay);
  return jitter ? Math.random() * exponential : exponential;
}

// Attempt 0 → 0-1000ms    Attempt 1 → 0-2000ms
// Attempt 2 → 0-4000ms    Attempt 3 → 0-8000ms
// Attempt 4 → 0-16000ms   Attempt 5+ → 0-30000ms (capped)

3. Retry-After Header — Server-Guided Backoff

// 429 Too Many Requests often includes Retry-After header
// Two formats:
//   Retry-After: 120          → seconds to wait
//   Retry-After: Fri, 31 Dec 2024 23:59:59 GMT  → absolute date

async function fetchWithRetryAfter(url, options, maxRetries = 3) {
  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    const response = await fetch(url, options);

    if (response.status === 429) {
      const retryAfter = response.headers.get('Retry-After');
      let delayMs;

      if (retryAfter) {
        const seconds = parseInt(retryAfter, 10);
        delayMs = isNaN(seconds)
          ? new Date(retryAfter).getTime() - Date.now()
          : seconds * 1000;
      } else {
        delayMs = calculateBackoff(attempt);
      }

      await new Promise(r => setTimeout(r, delayMs));
      continue;
    }

    return response;
  }
  throw new Error('Max retries exceeded');
}

4. Idempotency Keys — Safe Mutation Retries

// GET requests are idempotent — safe to retry
// POST/PUT/DELETE are NOT — retrying may cause double-charge, double-create
// Solution: idempotency keys let server detect duplicate submissions

// Flow with idempotency key:
// 1. Client generates UUID before first attempt
// 2. Client sends: POST /orders { idempotency_key: "abc-123" }
// 3. Server processes, stores result keyed by "abc-123"
// 4. Network fails, client doesn't receive response
// 5. Client retries: POST /orders { idempotency_key: "abc-123" }
// 6. Server sees "abc-123" → returns cached result (no double-create)

async function safeMutation(url, data, maxRetries = 3) {
  const idempotencyKey = crypto.randomUUID();

  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    try {
      const res = await fetch(url, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Idempotency-Key': idempotencyKey,
        },
        body: JSON.stringify(data),
      });
      return await res.json();
    } catch (err) {
      if (attempt === maxRetries) throw err;
      await new Promise(r => setTimeout(r, calculateBackoff(attempt)));
    }
  }
}

1. Layered Error Handling Architecture

// Error handling should be layered, not scattered
//
// Layer 1: Network Layer (fetch wrapper)
//   → Classify errors, transform to typed errors
//   → Attach metadata (status, url, timing)
//
// Layer 2: API Client Layer
//   → Retry transient errors with backoff
//   → Handle 401 with token refresh
//   → Respect rate limits (429)
//
// Layer 3: Data Layer (React Query / SWR)
//   → Configure retry policies per query
//   → Manage stale-while-revalidate
//   → Expose error states to UI
//
// Layer 4: UI Layer (Error Boundaries + Components)
//   → Display user-friendly messages
//   → Provide retry actions
//   → Log to monitoring
// Layer 1: Typed Error Transformer
class ApiError extends Error {
  constructor(message, status, url, body) {
    super(message);
    this.status = status;
    this.url = url;
    this.body = body;
    this.isRetryable = [408, 429, 502, 503, 504].includes(status);
    this.isAuthError = [401, 403].includes(status);
  }
}

async function apiFetch(url, options) {
  try {
    const res = await fetch(url, options);
    if (!res.ok) {
      const body = await res.json().catch(() => null);
      throw new ApiError(body?.message ?? res.statusText, res.status, url, body);
    }
    return res;
  } catch (err) {
    if (err instanceof ApiError) throw err;
    throw new ApiError('Network Error', 0, url, null);
  }
}

2. React Query Retry Configuration

// Global retry configuration with smart defaults
const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      retry: (failureCount, error) => {
        // Never retry client errors (4xx except 429)
        if (error.status >= 400 && error.status < 500 && error.status !== 429) {
          return false;
        }
        return failureCount < 3;
      },
      retryDelay: (attempt, error) => {
        // Respect Retry-After header for 429
        if (error.status === 429 && error.retryAfter) {
          return error.retryAfter * 1000;
        }
        return Math.min(1000 * 2 ** attempt, 30000);
      },
    },
    mutations: {
      retry: 0,  // NEVER retry mutations by default — side effects!
    },
  },
});

FAANG Interview Questions

Q1: How would you design an error handling strategy for a React application that consumes 15+ microservices?
I'd implement a layered approach. Layer 1: a fetch wrapper that transforms all responses into typed errors (ApiError class) with status, URL, and retryable flag. Layer 2: an API client that handles retry logic — exponential backoff with jitter for transient errors (5xx, 429), immediate failure for client errors (4xx). Layer 3: React Query with global retry configuration that respects Retry-After headers. Layer 4: granular Error Boundaries around independent page sections so a recommendations service failure doesn't crash the checkout flow. Each microservice gets its own circuit breaker instance. I'd also add a global error interceptor that catches 401s and silently refreshes tokens before retrying.
Follow-up: What happens during a partial outage where 3 of 15 services are down?
Q2: Explain the thundering herd problem and how you'd prevent it in a frontend retry strategy.
When a server recovers from an outage, all clients retry simultaneously — this synchronized burst can immediately re-down the server. It's like 10,000 people refreshing at once when a concert ticket site comes back. Prevention: exponential backoff ensures retries space out logarithmically (1s, 2s, 4s, 8s). But even exponential backoff synchronizes clients that started at the same time. The solution is full jitter — instead of waiting exactly 4 seconds, wait a random time in [0, 4s]. AWS's research paper shows full jitter provides the best spread. In React Query, I configure retryDelay with randomization built in. For critical systems, I'd also add a circuit breaker so clients stop retrying entirely after N failures and only probe periodically.
Follow-up: How does React Query handle retry state — does it block the UI or happen in the background?
Q3: When is it safe to retry a POST request? What mechanism makes this safe?
POST requests are not idempotent by default — retrying can cause duplicate orders, double payments, etc. The mechanism that makes it safe is an idempotency key: a unique UUID generated client-side before the first attempt, sent as a header (Idempotency-Key). The server stores the result keyed by this UUID. If the client retries with the same key, the server returns the cached result without re-processing. Stripe uses this pattern for payment APIs. The key must be generated once per logical operation (not per retry attempt). For frontend implementation, I generate the key when the user clicks "Submit" and reuse it across all retry attempts for that action.
Follow-up: What happens if the server crashes between processing the request and storing the idempotency key?
Q4: How do you handle errors differently in mutations vs queries?
Queries (GET) are safe to retry aggressively — they're idempotent and have no side effects. I configure React Query with retry: 3 and exponential backoff for queries. Mutations are fundamentally different: retrying a failed POST could create duplicate resources. My default is retry: 0 for mutations. For mutations that need reliability, I use idempotency keys and only retry on network errors (not 4xx or 5xx). The UI approach differs too: for queries, I show stale data + a subtle refresh indicator during retries. For mutations, I show explicit progress states (submitting → failed → retry button) so the user stays in control of side effects.
Follow-up: What about optimistic updates that need to be rolled back on mutation failure?

1. Complete Retry Utility with All Features

async function fetchWithRetry(url, options = {}, config = {}) {
  const {
    maxRetries = 3,
    baseDelay = 1000,
    maxDelay = 30000,
    retryOn = [408, 429, 500, 502, 503, 504],
    onRetry = () => {},
  } = config;

  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    try {
      const response = await fetch(url, options);

      if (!response.ok) {
        if (retryOn.includes(response.status) && attempt < maxRetries) {
          let delay;

          // Respect server's Retry-After header
          if (response.status === 429) {
            const retryAfter = response.headers.get('Retry-After');
            delay = retryAfter ? parseInt(retryAfter) * 1000 : calculateBackoff(attempt);
          } else {
            delay = calculateBackoff(attempt, { baseDelay, maxDelay });
          }

          onRetry({ attempt, delay, status: response.status });
          await new Promise(r => setTimeout(r, delay));
          continue;
        }
        throw new ApiError(response.statusText, response.status, url);
      }

      return response;
    } catch (err) {
      if (err instanceof ApiError) throw err;
      // Network error (no response) — retry
      if (attempt < maxRetries) {
        const delay = calculateBackoff(attempt, { baseDelay, maxDelay });
        onRetry({ attempt, delay, status: 0 });
        await new Promise(r => setTimeout(r, delay));
        continue;
      }
      throw err;
    }
  }
}

2. React Hook: useRetryableQuery

function useRetryableQuery(key, fetchFn, options = {}) {
  const [retryCount, setRetryCount] = useState(0);
  const [nextRetryAt, setNextRetryAt] = useState(null);

  return useQuery({
    queryKey: key,
    queryFn: fetchFn,
    retry: (failureCount, error) => {
      if (error.status >= 400 && error.status < 500 && error.status !== 429) {
        return false;
      }
      setRetryCount(failureCount);
      return failureCount < (options.maxRetries ?? 3);
    },
    retryDelay: (attempt, error) => {
      const delay = error.status === 429 && error.retryAfter
        ? error.retryAfter * 1000
        : Math.min(1000 * 2 ** attempt, 30000);
      setNextRetryAt(Date.now() + delay);
      return delay;
    },
    meta: { retryCount, nextRetryAt },
    ...options,
  });
}

3. Global 401 Interceptor with Token Refresh

let isRefreshing = false;
let refreshQueue = [];

async function authFetch(url, options = {}) {
  let response = await fetch(url, {
    ...options,
    headers: { ...options.headers, Authorization: `Bearer ${getAccessToken()}` },
  });

  if (response.status === 401) {
    if (!isRefreshing) {
      isRefreshing = true;
      try {
        await refreshToken();
        refreshQueue.forEach(cb => cb());
      } catch {
        redirectToLogin();
        return;
      } finally {
        isRefreshing = false;
        refreshQueue = [];
      }
    } else {
      await new Promise(resolve => refreshQueue.push(resolve));
    }
    // Retry original request with new token
    response = await fetch(url, {
      ...options,
      headers: { ...options.headers, Authorization: `Bearer ${getAccessToken()}` },
    });
  }

  return response;
}

Retry Strategy Decision Tree

Is it a GET request?
├── Yes → Retry up to 3x with exponential backoff + jitter
└── No (POST/PUT/DELETE)
    ├── Do you have an idempotency key?
    │   ├── Yes → Safe to retry (same key = same result)
    │   └── No → Do NOT retry (risk of duplication)
    └── Is it a 429?
        ├── Yes → Wait for Retry-After header value
        └── No → Fail and let user decide to retry

Backoff Formula

delay = min(baseDelay × 2^attempt, maxDelay) × random()

Default:  base=1000ms  max=30000ms  jitter=full

Attempt 0 → 0–1s     Attempt 1 → 0–2s
Attempt 2 → 0–4s     Attempt 3 → 0–8s
Attempt 4 → 0–16s    Attempt 5+ → 0–30s (capped)

Production Checklist

✅ Classify every error (transient vs permanent) before acting
✅ NEVER retry mutations without idempotency keys
✅ Use exponential backoff with FULL JITTER (not decorrelated)
✅ Respect Retry-After header for 429 responses
✅ Set a max retry count (3 for queries, 0 for mutations)
✅ Cap backoff delay (30s) to prevent absurd wait times
✅ Intercept 401 globally → silent token refresh → retry
✅ Use granular Error Boundaries (not one per app)
✅ Log retries to monitoring (track retry storms)
✅ Pair retry with circuit breaker for sustained outages
