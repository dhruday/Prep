Tenant context via AsyncLocalStorage

import { AsyncLocalStorage } from 'async_hooks';
export const tenantCtx = new AsyncLocalStorage<{ tenantId:string }>();

// Middleware — extract from JWT
export const tenantMiddleware = (req, res, next) => {
  const { tenantId } = req.jwtPayload; // from verified JWT
  tenantCtx.run({ tenantId }, next);  // available throughout request
};

// Repository auto-scopes every query
class TenantRepository<T> {
  findAll(where = {}) {
    const { tenantId } = tenantCtx.getStore()!;
    return this.repo.find({ where: { ...where, tenantId } });
  }
}

// Cache — tenant-namespaced keys
const cacheKey = (resource, id) => {
  const { tenantId } = tenantCtx.getStore()!;
  return `${tenantId}:${resource}:${id}`; // never cross-tenant cache hit
};
