Best practice: RBAC as the gateway (coarse-grained role check on endpoint), ABAC as the filter (resource-level predicate check). The combination handles 99% of real-world use cases.

ABAC policy engine in TypeScript

interface Policy<S=any,R=any,E=any> {
  effect: 'allow' | 'deny';
  actions: string[];
  condition: (subject:S, resource:R, env:E) => boolean;
}

class ABACEngine {
  private policies: Policy[] = [];

  addPolicy(p: Policy) { this.policies.push(p); }

  evaluate(sub:any, res:any, action:string, env:any = {}): 'allow'|'deny' {
    const applicable = this.policies.filter(p => p.actions.includes(action) || p.actions.includes('*'));
    // Deny-override: any deny beats all allows
    if (applicable.some(p => p.effect==='deny' && p.condition(sub,res,env))) return 'deny';
    return applicable.some(p => p.effect==='allow' && p.condition(sub,res,env)) ? 'allow' : 'deny';
  }
}

const engine = new ABACEngine();
engine.addPolicy({ effect:'allow', actions:['document:read'],
  condition:(sub,res) => sub.department === res.department });
engine.addPolicy({ effect:'deny', actions:['*'],
  condition:(sub,res) => res.classification === 'secret' && sub.clearanceLevel < 3 });
