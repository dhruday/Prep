Feature gates: DOM surgery, not CSS bandages

❌ display:none (insecure)

// Removed from DOM entirely
// No keyboard access
// No DevTools visibility

Permanently unavailable (wrong role) → HIDE from DOM completely
Plan/tier gate (upgrade needed) → DISABLE + show upgrade tooltip

Angular structural directive — *appCan

@Directive({ selector: '[appCan]', standalone: true })
export class CanDirective implements OnInit, OnDestroy {
  @Input('appCan') permission!: string;
  private sub?: Subscription;

  constructor(
    private tpl: TemplateRef<any>,
    private vcr: ViewContainerRef,
    private auth: AuthService
  ) {}

  ngOnInit() {
    this.sub = this.auth.permissions$.subscribe(perms => {
      this.vcr.clear();
      if (perms.has(this.permission))
        this.vcr.createEmbeddedView(this.tpl);
    });
  }

  ngOnDestroy() { this.sub?.unsubscribe(); }
}

React <Can> with fallback for plan-gate UX

const Can = ({ action, subject, children, fallback=null }) => {
  const ability = useAbility(AbilityContext);
  return ability.can(action, subject) ? children : fallback;
};

// Hide entirely:
<Can action="delete" subject="Invoice">
  <DeleteButton />
</Can>

// Disable with upgrade tooltip:
<Can action="export" subject="Report"
  fallback={<Tooltip text="Upgrade to Pro"><Button disabled /></Tooltip>}>
  <ExportButton />
</Can>
