Geo-based delivery: route users to the closest data center — reduce latency, comply with data residency laws

Geo delivery has two goals: (1) performance — closest PoP serves assets faster, (2) compliance — GDPR requires EU user data stays in EU. Implemented via GeoDNS (route DNS response to nearest region) or CDN edge rules (Cloudflare Route / AWS CloudFront Geo Restriction). In the frontend, geo context can also drive: locale detection, currency display, region-specific compliance text, and which features are available (e.g., some features legally cannot be shown in certain countries).

Cloudflare geo header + Next.js geo routing

// ─── Cloudflare Worker: read geo ───
export default {
  async fetch(req: Request, env: Env) {
    const country = (req as any).cf?.country ?? 'US';
    const euCountries = ['DE', 'FR', 'IT', 'ES', 'NL'];
    
    if (euCountries.includes(country)) {
      // rewrite to EU origin to ensure data residency
      return fetch(`https://eu.api.app.com${new URL(req.url).pathname}`);
    }
    return fetch(req);
  }
};

// ─── Next.js: locale from geo, user pref overrides ───
export function middleware(req: NextRequest) {
  const country = req.geo?.country ?? 'US';
  const preferredLocale = req.cookies.get('NEXT_LOCALE')?.value;
  const locale = preferredLocale ?? geoToLocale[country] ?? 'en';
  return NextResponse.rewrite(new URL(`/${locale}${req.nextUrl.pathname}`, req.url));
}

const geoToLocale: Record<string, string> = {
  DE: 'de', JP: 'ja', FR: 'fr', SA: 'ar'
};
