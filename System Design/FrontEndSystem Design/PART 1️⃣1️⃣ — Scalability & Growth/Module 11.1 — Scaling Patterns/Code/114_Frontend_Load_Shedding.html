Load shedding: the frontend deliberately refuses or degrades work when the system is overloaded

A frontend that fans out 20 API calls per user visit will destroy your backend at 1M DAU. Load shedding means the client cooperates in protecting the system: rate-limiting its own requests, short-circuiting redundant calls, batching small requests, and gracefully degrading features instead of hammering a struggling API. This is distinct from backend rate-limiting — it happens before the request is ever sent.

Client circuit breaker + request deduplication

// ─── Client circuit breaker ───
class CircuitBreaker {
  private failures = 0;
  private state: 'closed' | 'open' | 'half-open' = 'closed';
  private lastFailure = 0;

  async call<T>(fn: () => Promise<T>): Promise<T> {
    if (this.state === 'open') {
      if (Date.now() - this.lastFailure < 5000) throw new Error('Circuit open');
      this.state = 'half-open';
    }
    try {
      const result = await fn();
      this.failures = 0; this.state = 'closed';
      return result;
    } catch (err) {
      this.failures++;
      if (this.failures >= 3) { this.state = 'open'; this.lastFailure = Date.now(); }
      throw err;
    }
  }
}

// ─── Request deduplication ───
const inflight = new Map<string, Promise<any>>();
function deduplicatedFetch(url: string) {
  if (inflight.has(url)) return inflight.get(url)!;
  const p = fetch(url).finally(() => inflight.delete(url));
  inflight.set(url, p);
  return p;
}
