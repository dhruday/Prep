Data leaks from the frontend — more ways than you think

Frontend apps handle sensitive data constantly. Leaks occur through console logs, error tracking, analytics, URL parameters, clipboard, screenshot tools, browser extensions, autofill, and even CSS. Each vector is a side channel. GDPR fines apply even for unintentional leaks — SAP had dedicated data leakage assessment teams for all BTP products.

Production logger + Sentry scrubber + URL sanitizer

// ─── Production logger: no-op in prod ───
const logger = {
  log: (...a: any[]) => { if (process.env.NODE_ENV !== 'production') console.log(...a); },
  warn: (...a: any[]) => { if (process.env.NODE_ENV !== 'production') console.warn(...a); },
  error: (...a: any[]) => console.error(...a) // errors always logged but scrubbed
};

// ─── Sentry beforeSend: remove PII from error events ───
Sentry.init({
  beforeSend(event) {
    const sensitiveKeys = ['password','token','ssn','creditCard','email'];
    function scrub(obj: any): any {
      if (typeof obj !== 'object' || !obj) return obj;
      return Object.fromEntries(
        Object.entries(obj).map(([k,v]) =>
          [k, sensitiveKeys.includes(k.toLowerCase()) ? '[REDACTED]' : scrub(v)]
        )
      );
    }
    event.extra = scrub(event.extra);
    return event;
  }
});

// ─── Analytics: sanitize URL before sending ───
function sanitizeUrl(url: string): string {
  const u = new URL(url, window.location.origin);
  ['token','code','state','email'].forEach(p => u.searchParams.delete(p));
  return u.toString();
}
gtag('event','page_view',{page_location: sanitizeUrl(location.href)});

// ─── Strip auth code from URL after OAuth callback ───
const params = new URLSearchParams(location.search);
if (params.has('code')) {
  history.replaceState({}, '', location.pathname); // remove ?code= from URL
}
