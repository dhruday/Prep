Why keyboard navigation is critical

~26 million Americans have a motor disability limiting mouse use. Screen reader users navigate predominantly by keyboard. Power users (developers, traders) prefer keyboard. WCAG 2.1.1 requires all functionality available via keyboard, 2.1.2 prohibits keyboard traps. Adobe, Microsoft, and Salesforce all have keyboard nav in their a11y checklists.

Tab order
Should follow visual / logical reading order. Use tabIndex="-1" to remove, tabIndex="0" to add to flow. Never use positive tabIndex — breaks natural order.
Focus ring
NEVER suppress :focus outline globally. Use :focus-visible for mouse users (only shows ring on keyboard nav). Must be clearly visible with 3:1 contrast ratio (WCAG 2.4.11).
Focus management
On modal open: move focus inside. On modal close: return focus to trigger. On route change: move focus to h1 or to the new content area.

// ─── Focus trap hook ───
function useFocusTrap(active) {
  const containerRef = useRef(null);
  useEffect(() => {
    if (!active) return;
    const focusable = containerRef.current.querySelectorAll(
      'button,a,input,select,textarea,[tabindex]:not([tabindex="-1"])'
    );
    const first = focusable[0];
    const last  = focusable[focusable.length - 1];
    first?.focus();
    const onKey = e => {
      if (e.key !== 'Tab') return;
      if (e.shiftKey && document.activeElement === first) {
        e.preventDefault(); last.focus();
      } else if (!e.shiftKey && document.activeElement === last) {
        e.preventDefault(); first.focus();
      }
    };
    document.addEventListener('keydown', onKey);
    return () => document.removeEventListener('keydown', onKey);
  }, [active]);
  return containerRef;
}

// ─── Route change focus management (React Router v6) ───
function RouteChangeAnnouncer() {
  const location = useLocation();
  useEffect(() => {
    const h1 = document.querySelector('h1');
    h1?.focus();           // h1 needs tabIndex="-1" to be programmatically focusable
  }, [location]);
  return null;
}
