Well-designed component (TypeScript)

// ✅ Single Responsibility · Typed Props · No side effects in render
interface UserCardProps {
  name: string;
  role: string;
  avatarUrl: string;
  onViewProfile: (userId: string) => void;
}

const UserCard: React.FC<UserCardProps> = ({ name, role, avatarUrl, onViewProfile }) => (
  <div className="user-card">
    <img src={avatarUrl} alt={name} />
    <h3>{name}</h3>
    <p>{role}</p>
    <button onClick={() => onViewProfile(name)}>View Profile</button>
  </div>
);

// ❌ Anti-pattern — does too much
