Atomic folder structure

src/
  components/
    atoms/
      Button/           // Button.tsx + Button.stories.tsx + Button.test.tsx
      Input/
      Label/
      Icon/
    molecules/
      SearchBar/        // = Label + Input + Button
      FormField/        // = Label + Input + ErrorText
    organisms/
      NavigationHeader/ // = Logo + SearchBar + UserMenu
      UserTable/        // = Table + Pagination + FilterBar
    templates/
      DashboardLayout/  // = Header + Sidebar + ContentSlot (no real data)
    pages/
      DashboardPage/    // = DashboardLayout with real API data
