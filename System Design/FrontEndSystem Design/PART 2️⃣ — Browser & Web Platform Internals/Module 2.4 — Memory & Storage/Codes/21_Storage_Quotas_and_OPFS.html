📋 Production Pattern — Storage Guard

// Service Worker: request persistent storage on install
self.addEventListener('install', async (event) => {
  event.waitUntil(async function() {
    // 1. Cache app shell
    const cache = await caches.open('app-v3');
    await cache.addAll(['/index.html', '/app.js']);
    
    // 2. Request persistent storage
    const isPersistent = await navigator.storage.persist();
    console.log('Persistent:', isPersistent);
    
    // 3. Monitor usage — proactively prune at 80%
    const { usage, quota } = await navigator.storage.estimate();
    const pct = usage / quota;
    if (pct > 0.8) {
      // Delete old version caches
      const all = await caches.keys();
      await Promise.all(all.filter(n => n !== 'app-v3').map(n => caches.delete(n)));
    }
  }());
});

🔬 OPFS Sync Access Handle (Web Worker Pattern)

// ONLY available inside Web Workers — synchronous, high-performance
// Used by SQLite WASM, Photoshop Web, Adobe Acrobat Web

// web-worker.js
async function runInWorker() {
  const root = await navigator.storage.getDirectory();
  const fileHandle = await root.getFileHandle('data.db', { create: true });
  
  // Sync Access Handle — no await needed!
  const sync = await fileHandle.createSyncAccessHandle();
  
  const encoder = new TextEncoder();
  const data = encoder.encode(JSON.stringify({ records: [] }));
  
  sync.write(data, { at: 0 });      // sync write at byte offset 0
  sync.flush();                      // ensure bytes are persisted
  
  const readBuf = new Uint8Array(sync.getSize());
  sync.read(readBuf, { at: 0 });    // sync read
  sync.close();
  
  // Post result back to main thread
  self.postMessage({ data: new TextDecoder().decode(readBuf) });
}

⚡ Decision Guide
• Small config/preferences ({"<"}100KB): localStorage
• Structured app data (user data, cached API responses): IndexedDB + React Query / Dexie.js
• Offline network request caching (service worker): Cache API
• Large binary files, SQLite DB, WASM assets: OPFS
