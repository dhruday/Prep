1. Feature Tier Classification

// Before you can degrade gracefully, classify every feature:
//
// CRITICAL — App is unusable without this
//   Product details on product page, checkout cart, auth
//   → Never degrade. Always show error with retry option.
//
// CORE — Primary user workflow, can use fallback
//   Product images, inventory status, user profile
//   → Degrade to cached data, placeholder images, "check availability"
//
// ENRICHMENT — Improves experience, not essential
//   Reviews, "You may also like", product Q&A
//   → Show "temporarily unavailable" or hide section quietly
//
// OPTIONAL — Personalization, analytics, non-essential
//   Recommendations, tracking events, A/B test data
//   → Fail silently. User never notices.
// E-commerce product page example:
const featureMap = {
  'product-details':      'CRITICAL',     // Can't shop without this
  'product-images':       'CORE',         // Placeholder image if fails
  'inventory-status':     'CORE',         // "Check with seller" fallback
  'reviews':              'ENRICHMENT',   // Hide if unavailable
  'recommendations':      'ENRICHMENT',   // Hide "You may also like"
  'personalized-pricing': 'OPTIONAL',     // Show standard price
  'analytics-events':     'OPTIONAL',     // Drop silently
};

2. Promise.allSettled — The Key API

// Promise.all → rejects if ANY promise rejects (all or nothing)
// Promise.allSettled → waits for ALL, never rejects
// This is the foundation of graceful degradation

const results = await Promise.allSettled([
  fetchProductDetails(id),      // CRITICAL
  fetchProductImages(id),       // CORE
  fetchReviews(id),             // ENRICHMENT
  fetchRecommendations(userId), // OPTIONAL
]);

// Each result is either:
// { status: 'fulfilled', value: data }
// { status: 'rejected', reason: error }

// Process each independently:
const [details, images, reviews, recs] = results;

if (details.status === 'rejected') {
  // CRITICAL failed → show full error page
  throw details.reason;
}

// CORE: use fallback if failed
const productImages = images.status === 'fulfilled'
  ? images.value
  : ['/images/placeholder-product.png'];

// ENRICHMENT: null triggers "temporarily unavailable" in UI
const productReviews = reviews.status === 'fulfilled' ? reviews.value : null;

// OPTIONAL: null hides the section entirely
const productRecs = recs.status === 'fulfilled' ? recs.value : null;

3. Fallback Chain Pattern

// Try multiple sources before giving up:
//
// 1. Live API → freshest data
// 2. Stale cache → data from last successful fetch
// 3. Local storage → persisted across sessions
// 4. Static fallback → pre-built default content
// 5. Error state → show user-friendly error

async function fetchWithFallbackChain(key, liveFetcher, staticFallback) {
  // 1. Try live API
  try {
    const data = await liveFetcher();
    saveToCache(key, data);       // Update cache for future fallback
    return { data, source: 'live', stale: false };
  } catch (e) { /* fall through */ }

  // 2. Try stale cache
  const cached = getFromCache(key);
  if (cached) {
    return { data: cached.data, source: 'cache', stale: true, age: Date.now() - cached.ts };
  }

  // 3. Try localStorage (survives refresh)
  const stored = localStorage.getItem(`fallback:${key}`);
  if (stored) {
    return { data: JSON.parse(stored), source: 'localStorage', stale: true };
  }

  // 4. Static fallback (baked into app bundle)
  if (staticFallback) {
    return { data: staticFallback, source: 'static', stale: true };
  }

  // 5. Everything failed
  return { data: null, source: 'none', stale: false };
}

4. Network-Aware Degradation

// Use Network Information API to proactively degrade on slow connections

function getNetworkTier() {
  const conn = navigator.connection;
  if (!conn) return 'unknown';

  if (conn.effectiveType === 'slow-2g' || conn.effectiveType === '2g') return 'minimal';
  if (conn.effectiveType === '3g' || conn.saveData) return 'reduced';
  return 'full';
}

// Adjust what we fetch based on network quality:
const tier = getNetworkTier();

if (tier === 'minimal') {
  // Only fetch CRITICAL data, skip everything else
  fetchProductDetails(id);
} else if (tier === 'reduced') {
  // CRITICAL + CORE, skip ENRICHMENT and OPTIONAL
  Promise.allSettled([fetchProductDetails(id), fetchProductImages(id)]);
} else {
  // Full experience
  Promise.allSettled([fetchProductDetails(id), fetchProductImages(id),
    fetchReviews(id), fetchRecommendations(userId)]);
}

1. DegradedSection React Component Pattern

function DegradedSection({ data, isStale, tier, children }) {
  // OPTIONAL tier: hide completely if no data
  if (data === null && tier === 'OPTIONAL') return null;

  // ENRICHMENT: friendly notice
  if (data === null && tier === 'ENRICHMENT') {
    return (
      <div className="section-unavailable">
        This section is temporarily unavailable.
      </div>
    );
  }

  // CRITICAL with no data: full error
  if (data === null && tier === 'CRITICAL') {
    return <FullPageError message="Failed to load. Please retry." />;
  }

  // Has data (possibly stale) — render with stale indicator
  return (
    <div>
      {isStale && <div className="stale-banner">Showing cached data</div>}
      {children(data)}
    </div>
  );
}

3. Stale-While-Revalidate Strategy

// SWR: show stale data immediately, refresh in background
// Best UX for degradation: user sees "something" instantly

const { data, isFetching, isStale } = useQuery({
  queryKey: ['products'],
  queryFn: fetchProducts,
  staleTime: 5 * 60 * 1000,      // Fresh for 5 min

});

// Timeline:
// t=0:    First load → skeleton → fetch → show data
// t=3min: Re-render → data still fresh → show immediately (no fetch)
// t=6min: Re-render → data stale → show stale + background fetch
// t=6.5min: Fetch completes → seamlessly update to fresh data
// t=35min: Cache expired → skeleton → fetch → show data

FAANG Interview Questions

Q1: How would you design a frontend that remains usable when 3 out of 10 microservices are down?
I'd start by classifying every feature into tiers: CRITICAL (auth, core data), CORE (enrichment with fallbacks), ENRICHMENT (hide gracefully), OPTIONAL (fail silently). Then I'd use Promise.allSettled for page loads — not Promise.all — so each service fails independently. Each section gets a DegradedSection wrapper that renders appropriately for its tier. Circuit breakers per service detect outages fast. For CORE features, I maintain a fallback chain: live API → stale cache → localStorage → static defaults. For ENRICHMENT, null data triggers "temporarily unavailable" text. For OPTIONAL, the section disappears entirely. The user gets 70% of the page instantly, not a blank screen with a spinner.
Follow-up: How do you communicate to users which parts of the page are degraded?
Q2: What's the difference between graceful degradation and progressive enhancement?
They're complementary approaches from opposite directions. Graceful degradation starts with the full experience and removes features when things fail — you design the Porsche first, then figure out how to make it drive with a flat tire. Progressive enhancement starts with the minimum viable experience and adds features when capabilities are available — you design the bicycle first, then add the engine. For API failures, graceful degradation is the appropriate model: you designed for 10 services being up, and you define what happens when some go down. Progressive enhancement is more about browser capabilities: start with semantic HTML, enhance with CSS, enhance with JavaScript. In practice, good frontend architecture uses both: progressive enhancement for browser features, graceful degradation for service reliability.
Follow-up: How does this apply to mobile / low-bandwidth users?
Q3: How would you implement a fallback chain for product images?
The chain: 1) CDN image URL from the API response (primary). 2) Cached image URL from service worker / browser cache (offline support). 3) Thumbnail/lower-resolution variant (reduced quality, faster load). 4) CSS placeholder with product initials (no network, instant). 5) Generic category placeholder (electronics icon, clothing icon). I implement this with the img element's onerror handler cascading through sources, or with a React component that manages a source array and falls back on error. The key insight is that each fallback should be strictly "closer to the client" than the previous one — CDN → browser cache → bundled asset → generated CSS. This means each step is faster and more reliable than the last.
Follow-up: How does lazy loading interact with the fallback chain?
Q4: How do you prevent layout shifts when sections degrade?
CLS (Cumulative Layout Shift) is the main risk with degradation. When a section goes from loading to "temporarily unavailable" to hidden, the layout can jump if dimensions aren't reserved. My approach: 1) Every degradable section has a fixed minimum height matching its loaded state. 2) When a section is hidden (OPTIONAL tier), it uses display:none with a smooth CSS transition. 3) Skeleton loaders match the exact dimensions of the real content. 4) If an ENRICHMENT section shows "temporarily unavailable," that message occupies the same space as the loaded content. I test CLS by simulating random API failures with Playwright and measuring CLS scores — any degradation scenario should keep CLS under 0.1.
Follow-up: How would you handle degradation for dynamically-sized content like reviews?

1. Degradation Orchestrator

class DegradationOrchestrator {
  cache = new Map();

  async execute(key, strategy) {
    // 1. Try primary source
    try {
      const data = await strategy.primary();
      this.cache.set(key, { data, ts: Date.now() });
      return { data, source: 'live', stale: false };
    } catch (e) {
      console.warn(`Primary failed for ${key}:`, e.message);
    }

    // 2. Try stale cache
    const cached = this.cache.get(key);
    if (cached && Date.now() - cached.ts < (strategy.maxStaleAge ?? 300000)) {
      return { data: cached.data, source: 'cache', stale: true };
    }

    // 3. Try fallbacks in order
    for (const fb of (strategy.fallbacks ?? [])) {
      try {
        const data = await fb.execute();
        return { data, source: fb.name, stale: false };
      } catch (e) {
        console.warn(`Fallback ${fb.name} failed`);
      }
    }

    // 4. All sources failed
    return { data: null, source: 'none', stale: false };
  }
}

// Usage:
const orchestrator = new DegradationOrchestrator();

const result = await orchestrator.execute('products', {
  primary: () => fetch('/api/products').then(r => r.json()),
  maxStaleAge: 5 * 60 * 1000, // Accept 5-min-old cache
  fallbacks: [
    { name: 'localStorage', execute: () => JSON.parse(localStorage.products) },
    { name: 'static', execute: () => import('./fallback-products.json') },
  ],
});

2. Network-Aware Data Loading Hook

function useAdaptiveQuery(queryKey, fetchFn, tier) {
  const networkTier = useNetworkTier(); // 'full' | 'reduced' | 'minimal'

  // Skip non-essential fetches on slow connections
  const enabled = networkTier === 'full'
    || tier === 'CRITICAL'
    || (tier === 'CORE' && networkTier !== 'minimal');

  return useQuery({
    queryKey,
    queryFn: fetchFn,
    enabled,
    staleTime: networkTier === 'full' ? 5 * 60000 : 15 * 60000,
    placeholderData: getCachedFallback(queryKey),
  });
}
