Monitoring Integration

class MonitoredBreaker extends CircuitBreaker {
  constructor(name, opts) {
    super(opts);
    this.name = name;
    this.totalCalls = 0;
    this.totalSuccesses = 0;
    this.totalFailures = 0;
    this.lastStateChange = null;
  }

  get successRate() {
    if (!this.totalCalls) return 100;
    return ((this.totalSuccesses / this.totalCalls) * 100).toFixed(1);
  }

  onStateChange(prev, next) {
    this.lastStateChange = new Date();
    // emit metric / log to observability platform
  }
}
