State Colocation Principle (Kent C. Dodds)

// "Place state as close to where it's relevant as possible"
// — KCD's State Colocation blog post
//
// State placement spectrum:
//
//  Local ←─────────────────────────────────────→ Global
//  useState    useReducer   Context    Zustand    Redux
//  (component) (component)  (subtree)  (global)   (global)
//
// Rule: Start LOCAL. Only lift when multiple components need it.
//
// Decision tree:
// Q: Does ONLY this component use this state?
//    → YES: useState in that component ✅
//    → NO: Do sibling components need it?
//          → YES: Lift to shared parent (lowest common ancestor)
//          → NO, but deep children need it?
//                → Context (or composition pattern)
//          → Accessed from many unrelated parts?
//                → Global store (Zustand/Redux)

Composition Pattern (Children as Props)

// ❌ Everything re-renders when `open` changes
function Layout() {
  const [open, setOpen] = useState(false);
  return (
    <div>
      <SidePanel isOpen={open} />
      <ExpensiveTree /> // re-renders! No connection to `open`
    </div>
  );
}

// ✅ Extract state into its own component, pass children
function SidePanelWrapper({ children }) {
  const [open, setOpen] = useState(false);
  return (
    <div>
      <SidePanel isOpen={open} />
      {children} // children is a stable reference from parent!
    </div>
  );
}

function Layout() {
  return (
    <SidePanelWrapper>
      <ExpensiveTree /> // ✅ Does NOT re-render when panel toggles
    </SidePanelWrapper>
  );
}

// Why? Because <ExpensiveTree /> is created in Layout's render.
// Layout doesn't re-render → same JSX element → same reference.
// This is the "moving state down" + "composition" pattern.
// React team recommends this BEFORE reaching for memo.

Context Splitting to Prevent Re-renders

// ❌ Single context with multiple values
const AppContext = createContext();

function AppProvider({ children }) {
  const [user, setUser] = useState(null);
  const [theme, setTheme] = useState('dark');
  const [locale, setLocale] = useState('en');
  
  // ❌ Every consumer re-renders when ANY value changes
  return (
    <AppContext.Provider value={{ user, theme, locale, setUser, setTheme }}>
      {children}
    </AppContext.Provider>
  );
}

// ✅ Split into focused contexts
const UserContext = createContext();
const ThemeContext = createContext();
const LocaleContext = createContext();

// Now ThemeToggle only re-renders when theme changes
// UserMenu only re-renders when user changes

// ✅ Alternative: use-context-selector library
import { createContext, useContextSelector } from 'use-context-selector';

const theme = useContextSelector(AppContext, ctx => ctx.theme);
// Only re-renders when ctx.theme changes!
