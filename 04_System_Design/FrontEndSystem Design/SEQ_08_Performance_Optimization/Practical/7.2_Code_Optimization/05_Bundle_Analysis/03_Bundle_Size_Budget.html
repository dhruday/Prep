Webpack Performance Budgets

// webpack.config.js
module.exports = {
  performance: {
    hints: 'error',             // 'warning' | 'error' | false
    maxEntrypointSize: 250000,   // 250KB for initial load
    maxAssetSize: 200000,        // 200KB per individual file
  },
};

// This will fail the build if any JS chunk exceeds 200KB
// or if the total initial entrypoint exceeds 250KB

// Per-chunk budgets (granular control):
optimization: {
  splitChunks: {
    maxSize: 150000, // 150KB — split chunks exceeding this
    cacheGroups: {
      vendor: {
        test: /[\\/]node_modules[\\/]/,
        name: 'vendor',
        maxSize: 200000, // 200KB for vendor chunk
      },
    },
  },
},

bundlesize / Size Limit (CI Integration)

// package.json — bundlesize (simpler)
{
  "bundlesize": [
    { "path": "dist/main.*.js", "maxSize": "170 kB" },
    { "path": "dist/vendor.*.js", "maxSize": "120 kB" },
    { "path": "dist/*.css", "maxSize": "30 kB" }
  ]
}

// package.json — size-limit (more accurate, by @ai/size-limit)
{
  "size-limit": [
    {
      "path": "dist/index.js",
      "limit": "170 KB",
      "gzip": true,         // measure gzipped
      "running": false      // don't execute, just measure
    },
    {
      "name": "Critical CSS",
      "path": "dist/critical.css",
      "limit": "14 KB"
    }
  ],
  "scripts": {
    "size": "size-limit",
    "size:check": "size-limit --check" // exits non-zero if over budget
  }
}

# GitHub Action for PR size checks
- name: Check bundle size

Google's Recommended Budgets

// Google Web Performance team recommendations:
//
// Initial JavaScript:  ≤ 170KB compressed (< 500KB uncompressed)
//   → Ensures < 5s TTI on mid-tier mobile (Moto G4)
//
// Critical CSS:        ≤ 14KB compressed
//   → Fits in first TCP round trip (14KB initial congestion window)
//
// Total page weight:   ≤ 500KB compressed
//   → Achievable time-to-interactive target
//
// Per-route JS budget: ≤ 50KB compressed (for code-split routes)
//   → Fast route transitions
//
// Images:              Responsive, WebP/AVIF, lazy-loaded below fold
// Fonts:               ≤ 2 font files, font-display: swap
//
// Track with: Lighthouse CI → Performance budget assertions
