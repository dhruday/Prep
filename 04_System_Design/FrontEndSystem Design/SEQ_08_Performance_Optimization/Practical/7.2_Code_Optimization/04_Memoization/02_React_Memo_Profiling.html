React.memo: Shallow Comparison Bailout

// React.memo wraps a component to skip re-renders
// when props haven't changed (shallow equality)

const ExpensiveRow = React.memo(function ExpensiveRow({ data, onClick }) {
  console.log('ExpensiveRow rendered', data.id);
  return (
    <tr onClick={onClick}>
      <td>{data.name}</td>
      <td>{heavyFormat(data.value)}</td>
    </tr>
  );
});

// ⚠️ Common pitfall: inline objects break memo
// ❌ style={{color:'red'}} — creates new object every render
// ✅ const style = useMemo(() => ({color:'red'}), [])

Custom Comparison Function

// Second argument: custom areEqual function
const Chart = React.memo(
  function Chart({ data, theme }) {
    return <canvas ref={drawChart(data)} />;
  },
  (prevProps, nextProps) => {
    // Return true to SKIP re-render (areEqual → true)
    // Only re-render when data changes, ignore theme
    return prevProps.data === nextProps.data;
  }
);

// ⚠️ Custom comparison is rarely needed
// Use it when:
// - You want to compare only specific props
// - Deep equality on a specific prop is required
// - Ignoring callback props that are known stable

  id,           // "Navigation"
