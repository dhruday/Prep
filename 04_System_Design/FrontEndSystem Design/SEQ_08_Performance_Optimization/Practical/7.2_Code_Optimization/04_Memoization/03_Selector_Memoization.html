Interactive: Memoized Selector vs Raw Selector

Reselect: createSelector Pattern

import { createSelector } from 'reselect';

// Input selectors (cheap — extract slices from state)
const selectItems = (state) => state.cart.items;
const selectTaxRate = (state) => state.config.taxRate;

// Memoized selector (expensive — only runs when inputs change)
const selectCartTotal = createSelector(
  [selectItems, selectTaxRate],     // input selectors
  (items, taxRate) => {             // result function
    console.log('Computing cart total...'); // only fires on change
    const subtotal = items.reduce((sum, i) => sum + i.price * i.qty, 0);
    return subtotal * (1 + taxRate);
  }
);

// Usage in component
function CartTotal() {
  const total = useSelector(selectCartTotal);
  return <span>${total.toFixed(2)}</span>;
}

// How it works:
// 1. Input selectors run on every dispatch
// 2. Results compared with === (referential equality)
// 3. If ALL inputs unchanged → return cached result
// 4. If ANY input changed → re-run result function

Parameterized Selectors (Per-Instance Memoization)

// ⚠️ Problem: shared selector across multiple component instances
const selectUserById = createSelector(
  [selectAllUsers, (_, userId) => userId],
  (users, userId) => users.find(u => u.id === userId)
);
// Cache size = 1! Different userId args thrash the cache.

// ✅ Solution 1: Factory pattern (one selector per instance)
const makeSelectUserById = () => createSelector(
  [selectAllUsers, (_, userId) => userId],
  (users, userId) => users.find(u => u.id === userId)
);

function UserCard({ userId }) {
  const selectUser = useMemo(makeSelectUserById, []); // one per mount
  const user = useSelector(state => selectUser(state, userId));
  ...
}

// ✅ Solution 2: re-reselect (Reselect v5+ has built-in)
import createCachedSelector from 're-reselect';

const selectUserById = createCachedSelector(
  [selectAllUsers, (_, userId) => userId],
  (users, userId) => users.find(u => u.id === userId)
)(
  (_, userId) => userId // cache key resolver
);

// ✅ Solution 3: Reselect v5 lruMemoize
import { createSelector, lruMemoize } from 'reselect';

const selectUserById = createSelector(
  [selectAllUsers, (_, userId) => userId],
  (users, userId) => users.find(u => u.id === userId),
  { memoize: lruMemoize, memoizeOptions: { maxSize: 50 } }
);

Zustand / Jotai / Signal Selectors

// Zustand: auto-memoized with shallow equality
import { useStore } from './store';
import { shallow } from 'zustand/shallow';

// ❌ Creates new object every time → always re-renders
const { name, age } = useStore(s => ({ name: s.name, age: s.age }));

// ✅ Use shallow compare for object selectors
const { name, age } = useStore(
  s => ({ name: s.name, age: s.age }),
  shallow
);

// ✅ Or select primitives individually (=== comparison)
const name = useStore(s => s.name);
const age = useStore(s => s.age);

// Jotai: derived atoms are automatically memoized
const totalAtom = atom(get => {
  const items = get(cartItemsAtom);
  return items.reduce((s, i) => s + i.price, 0);
}); // only recomputes when cartItemsAtom changes
