📐 Visual Stability
CLS — Cumulative Layout Shift

CLS >0.1 = poor
CLS <0.1 = good
Core Web Vitals (Google ranking signal): LCP · INP · CLS — these three directly affect SEO ranking. SAP Fiori apps improved LCP from 4.2s → 1.8s, CLS from 0.25 → 0.04 after optimization sprint.

Measuring metrics in code

// ─── web-vitals library (recommended) ───
import { onLCP, onINP, onCLS, onFCP, onTTFB } from 'web-vitals';

function sendToAnalytics(metric: Metric) {
  navigator.sendBeacon('/analytics', JSON.stringify({
    name:  metric.name,
    value: metric.value,
    id:    metric.id,
    page:  location.pathname,
  }));
}
onLCP(sendToAnalytics);
onINP(sendToAnalytics);
onCLS(sendToAnalytics);

// ─── PerformanceObserver (raw API) ───
const po = new PerformanceObserver((list) => {
  list.getEntries().forEach(entry => {
    if (entry.duration > 50) {  // long task threshold
      console.warn('Long task:', entry.duration.toFixed(1) + 'ms', entry.attribution);
    }
  });
});
po.observe({ type: 'longtask', buffered: true });

// ─── Layout Shift tracking ───
let cumulativeCLS = 0;
new PerformanceObserver(list => {
  list.getEntries().forEach(e => {
    if (!e.hadRecentInput) cumulativeCLS += e.value;
  });
}).observe({ type: 'layout-shift', buffered: true });
