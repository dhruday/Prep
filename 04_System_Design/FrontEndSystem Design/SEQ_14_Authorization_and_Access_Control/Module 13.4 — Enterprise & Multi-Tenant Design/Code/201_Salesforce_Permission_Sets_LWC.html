LWC — Custom Permission + FLS-aware rendering

import { LightningElement, api, wire } from 'lwc';
import { getRecord, getFieldValue } from 'lightning/uiRecordApi';
import hasApprovePermission from '@salesforce/customPermission/Approve_Invoice';
import AMOUNT from '@salesforce/schema/Invoice__c.Amount__c';
import NOTES from '@salesforce/schema/Invoice__c.Internal_Notes__c';

export default class InvoiceDetail extends LightningElement {
  @api recordId;

  @wire(getRecord, { recordId: '$recordId', fields: [AMOUNT, NOTES] })
  invoice;

  // FLS-aware: undefined if no access
  get amount() { return getFieldValue(this.invoice.data, AMOUNT); }
  get notes() { return getFieldValue(this.invoice.data, NOTES); }
  get showNotes() { return this.notes !== undefined; }

  // Custom Permission — compile-time constant
  get canApprove() { return hasApprovePermission; }
}

Apex — with sharing + FeatureManagement

public with sharing class InvoiceController {
  @AuraEnabled(cacheable=true)
  public static Invoice__c getInvoice(Id recordId) {
    return [SELECT Id, Amount__c, Status__c
            FROM Invoice__c WHERE Id = :recordId];
    // sharing rules automatically applied
  }

  @AuraEnabled
  public static void approveInvoice(Id recordId) {
    if (!FeatureManagement.checkPermission('Approve_Invoice'))
      throw new AuraHandledException('Permission denied');
    // ... approve logic
  }
}
