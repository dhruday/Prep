User A → User B data

Temporal
Access after expiry

Core rule
Server-side role assignment authority: you cannot assign a role higher than your own. Role comes from JWT (server-verified), never from client body. All changes create an audit record before taking effect.

Role assignment with authority boundary check

@Post('/users/:id/roles')
@UseGuards(JwtAuthGuard, RoleGuard('admin'))
async assignRole(
  @Param('id', ParseUUIDPipe) targetUserId: string,
  @Body() dto: AssignRoleDto,
  @CurrentUser() requestor: AuthUser
) {
  const targetRole = await this.roleService.findByName(dto.role);

  // BOUNDARY CHECK: cannot assign higher than yourself
  if (requestor.roleLevel < targetRole.level)
    throw new ForbiddenException('Cannot assign role exceeding own level');

  // AUDIT BEFORE CHANGE — immutable record
  await this.auditService.log({
    action: 'role:assign', targetUserId, role: dto.role,
    grantedBy: requestor.id, grantedAt: new Date()
  });

  await this.userService.assignRole(targetUserId, dto.role);
  await this.tokenService.revokeAllForUser(targetUserId); // force re-login
}
